"""Exact model of ingest.hl.WeightLimit (zero or given latency) and adversarial search of the T3 waiting bound."""
import random, sys, math
B = 120

def admit_time(spent, ready, w):
    """earliest t >= ready with sum(w_i for ts_i > t-60) + w <= B (matches 'now-t<60' retention). `spent` is a list
    of (ts, w) in nondecreasing ts; only the tail within 60 s of the last admission matters."""
    t = ready
    while True:
        live = [(ts, x) for ts, x in spent[-40:] if t - ts < 60]
        if sum(x for _, x in live) + w <= B:
            return t
        t = live[0][0] + 60

def run(weights, lat=0.0, start=0.0, pre=()):
    spent = list(pre); t = start; times = []
    for i, w in enumerate(weights):
        t = admit_time(spent, t, w); spent.append((t, w)); times.append(t)
        t += lat if not callable(lat) else lat(i)
    return t, times

def blocks(n_daily_el=278, n_daily_non=70, n_core=176, d=5):
    meta = [[20] * (d + 2)]
    el = [[39, 21, 21]] * n_daily_el
    non = [[21, 21]] * n_daily_non
    hr = [[100, 21, 21]] * n_core
    return meta, el, non, hr

if __name__ == "__main__":

    print("max over 300 random block orders", best)
    # structured: alternate hourly with daily-eligible, hourly first, daily first
