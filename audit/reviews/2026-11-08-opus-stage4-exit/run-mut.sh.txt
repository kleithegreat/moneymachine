#!/usr/bin/env bash
K=/tmp/rv-s4-opus/w/kernel; V=/tmp/rv-s4-opus/variants
go() { # variant canon rules args...
  v=$1; cp $V/$2 $K/mm/src/canon.rs; cp $V/$3 $K/mm/tests/rules.rs; touch $K/mm/src/canon.rs $K/mm/tests/rules.rs; shift 3
  (cd $K && nix develop /home/kevin/repos/mm-rv-s4 -c env CARGO_TARGET_DIR=/tmp/rv-s4-opus/tgt cargo test --release --locked -p mm "$@") > /tmp/rv-s4-opus/$v.log 2>&1
  echo exit=$? >> /tmp/rv-s4-opus/$v.log
}
go m1 m1.canon.rs orig.rules.rs --test canon --test keys --test rules -- --nocapture
go m2 m2.canon.rs orig.rules.rs --test canon --test keys --test rules -- --nocapture
go m3 m3.canon.rs orig.rules.rs --test canon --test keys --test rules -- --nocapture
go t1 orig.canon.rs t1.rules.rs --test rules -- every_rule_is_exact signed_zeros_and_undefined_conditions
go t2 orig.canon.rs t2.rules.rs --test rules -- every_rule_is_exact signed_zeros_and_undefined_conditions
echo done > /tmp/rv-s4-opus/mut.done
