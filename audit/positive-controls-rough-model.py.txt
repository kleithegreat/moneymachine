"""Paper PC2/PC4's rough model, NOT the kernel: the real N2 generator and frozen search-era funding (hash-checked; set
STORE to a store holding it), numpy positions, a flat 6.5 bps cost per unit turnover and the venue's $10 minimum order on
a fixed $10,000 equity. Magnitudes only. Usage: python audit/positive-controls-rough-model.py.txt <beta> <pairs>.
At beta .025 over 40 ten-year pairs: per-market net Sharpe SD .301 (raw) and .324 (raw24), plant Sharpe G SD .100 and
.076; G at 0.1/0.05/0.02 of size 1.834/1.834/1.832 (raw) and 1.182/1.177/1.150 (raw24), net Sharpe rising with the
smaller size (1.036 to 1.056, 1.009 to 1.014)."""
import json, os, sys
from pathlib import Path
import numpy as np
sys.path.insert(0, str(Path(__file__).resolve().parents[1]))
from gate import markets

spec = markets.null_params()["N2"]
spec["funding"]["store"] = os.environ.get("STORE", "data/store")
spec["funding"]["arrays"] = markets.frozen_funding(spec["funding"])
COST, EQ, MIN, A = 6.5e-4, 1e4, 10.0, 2190


def refs(f, n):
    ah = -np.clip((f - 1.25e-5) / 1e-4, -1, 1)
    idx = np.maximum(4 * np.arange(n) - 4, 0)
    a = ah[idx]; a[0] = 0
    cs = np.cumsum(np.vstack([np.zeros((1, ah.shape[1])), ah]), 0)
    m24 = (cs[idx + 1] - cs[np.maximum(idx + 1 - 24, 0)]) / np.minimum(idx + 1, 24)[:, None]; m24[0] = 0
    return {"raw": a, "raw24": m24}


def held(target):
    """Notional held under the minimum order: a change below MIN is skipped unless the target is exactly zero."""
    h = np.zeros_like(target); cur = np.zeros(target.shape[1])
    for k in range(target.shape[0]):
        t = target[k]; d = t - cur
        move = (t == 0) | (np.abs(d) >= MIN)
        cur = np.where(move, t, cur); h[k] = cur
    return h


def net(u, eps, scale, minimum=True):
    t = scale * u * EQ
    h = held(t) if minimum else t
    turn = np.abs(np.diff(np.vstack([np.zeros((1, h.shape[1])), h]), axis=0)).sum(1) / EQ
    return (h / EQ * eps).sum(1) - COST * turn


b, pairs = float(sys.argv[1]), int(sys.argv[2])
progs = [(k, s, m) for k in ("raw", "raw24") for s, m in ((0.1, False), (0.1, True), (0.05, True), (0.02, True))]
S = {p: [] for p in progs}; G = {p: [] for p in progs}
for i in range(pairs):
    z = markets.generate("N2", spec, f"rv3-t-{i}", 3650, (), 0.0)
    p = markets.generate("N2", spec, f"rv3-t-{i}", 3650, (0, 1), b)
    r = refs(z["funding"], z["eps"].shape[0])
    for key in progs:
        k, s, m = key
        rp, r0 = net(r[k], p["eps"], s, m), net(r[k], z["eps"], s, m)
        S[key].append(np.sqrt(A) * rp.mean() / rp.std()); G[key].append(np.sqrt(A) * (rp.mean() - r0.mean()) / rp.std())
out = {f"{k}@{s}{'' if m else ' linear'}": {"S_mean": float(np.mean(S[(k, s, m)])), "S_sd": float(np.std(S[(k, s, m)], ddof=1)),
       "G_mean": float(np.mean(G[(k, s, m)])), "G_sd": float(np.std(G[(k, s, m)], ddof=1))} for (k, s, m) in progs}
print(json.dumps({"beta": b, "pairs": pairs, **out}, indent=1))
