def yardstick() -> None:
    """The Gaussian fixed-horizon yardstick, z = S * sqrt(T): a planning reference, not a limit on every test."""
    print(f"z* = {Z80:.3f}, z*^2 = {Z80**2:.2f}")
    for sharpe in (1, 1.4, 2, 2.4, 5):
        print(f"  Sharpe {sharpe}: {Z80**2 / sharpe**2:.2f} years at 80% power")
    for years in (91 / 365, 1, 2):
        print(f"  T = {years:.3f} years: Sharpe {Z80 / math.sqrt(years):.2f} at 80% power")


def crossing(drift: float, spread: float) -> float:
    """P(a Brownian log wealth with total drift `drift` and total standard deviation `spread` reaches log 20
    at any time before the end): the reflection formula. The test stops at 20, so this is its power."""
    reflected = math.exp(min(0.0, 2 * drift * LOG20 / spread**2 + norm.logcdf((-drift - LOG20) / spread)))
    return float(norm.cdf((drift - LOG20) / spread)) + reflected


def kelly_power(u: float) -> float:
    """Power of an exact test staked at Kelly's fraction, where u = effect * sqrt(count)."""
    return crossing(u * u / 2, u)


U80 = brentq(lambda u: kelly_power(u) - 0.8, 1, 6)


def exact_cost() -> None:
    """Independent even bets with net edge e, staked at Kelly's fraction e: log wealth after N bets is near
    Gaussian with mean N*e^2/2 and standard deviation e*sqrt(N)."""
    z = norm.ppf(0.80)
    terminal = z + math.sqrt(z * z + 2 * LOG20)
    print(f"  stopping at 20: N = {U80**2:.2f} / e^2 for 80% power (u = {U80:.3f}; Argon wrote 8.67); the yardstick: {Z80**2:.2f} / e^2, so the exact test costs {U80**2 / Z80**2:.2f} times as much")
    print(f"  revision 3 asked for wealth of 20 at the end: N = {terminal**2:.1f} / e^2, where the power is really {kelly_power(terminal):.3f}")
    for edge in (0.03, 0.05, 0.08, 0.1):
        print(f"    net edge {edge}: {U80**2 / edge**2:,.0f} independent bets")
    print(f"  within 104 weeks: count a week * e^2 >= {U80**2 / 104:.4f}")
    for bets in (5, 10, 15, 33, 45, 60):
        edge = U80 / math.sqrt(104 * bets)
        print(f"    {bets} bets a week: net edge {edge:.3f} (a hit rate of {(1 + edge) / 2:.3f} on an even bet); power at 0.03, 0.05, 0.08: " + ", ".join(f"{kelly_power(e * math.sqrt(104 * bets)):.2f}" for e in (0.03, 0.05, 0.08)))
    for edge in (0.03, 0.05, 0.08):
        print(f"    net edge {edge}: {U80**2 / (104 * edge**2):.0f} bets a week")
    print(f"  discordant paired forecasts for a hit-rate gain of 0.05 (edge 0.1): {U80**2 / 0.1**2:.0f}; of 0.025: {U80**2 / 0.05**2:.0f}")


def account(bets: int, hold: int, edge: float, fraction: float, weeks: int = 104, paths: int = 4000) -> float:
    """A paper account in even bets with net edge `edge`: `bets` purchases a week, each staking `fraction` of the
    cash left at that moment and paid `hold` purchases later (0: before the next purchase). Returns P(cash reaches
    20 within `weeks`); a payout still pending at the end is not cash (Argon's round 4, finding 7)."""
    rng = np.random.default_rng(11)
    cash = np.ones(paths)
    pending = np.zeros((paths, hold + 1))
    reached = np.zeros(paths, dtype=bool)
    for bet in range(weeks * bets):
        slot = bet % (hold + 1)
        cash = cash + pending[:, slot]
        reached |= cash >= 20
        stake = fraction * cash
        cash = cash - stake
        pending[:, slot] = 2 * stake * (rng.random(paths) < (1 + edge) / 2)
    return float(reached.mean())


def forward_account() -> None:
    """The account's sequential rule simulated with three settlement timings: before the next purchase, a day
    later, and a week later (revision 3's batch, which Argon's finding 2 showed starves the stake)."""
    print(f"  no edge, 33 bets a week staking the cap, paid a day later: {account(33, 5, 0.0, STAKE_CAP, paths=20_000):.3f} of paths reach 20 (Ville's bound: 0.05 over any horizon)")
    print("  net edge for 80% power within 104 weeks, stake min(5%, Kelly) of cash:")
    for bets in (5, 10, 15, 33, 60):
        found = []
        for hold in (0, math.ceil(bets / 7), bets):
            found.append(brentq(lambda e: account(bets, hold, e, min(STAKE_CAP, e)) - 0.8, 0.01, 0.6, xtol=0.001))
        print(f"    {bets:2d} bets a week: paid at once {found[0]:.3f}, a day later {found[1]:.3f} (a hit rate of {(1 + found[1]) / 2:.3f}), a week later {found[2]:.3f}; closed form {U80 / math.sqrt(104 * bets):.3f}")
    for bets in (10, 33):
        hold = math.ceil(bets / 7)
        print(f"    {bets} bets a week paid a day later, power at 0.03, 0.05, 0.08: " + ", ".join(f"{account(bets, hold, e, min(STAKE_CAP, e)):.2f}" for e in (0.03, 0.05, 0.08)))


def capped_growth(mu: float, sigma: float, cap: float) -> float:
    """Expected log growth a step of wealth bet at the constant fraction `cap`, to second order."""
    return cap * mu - cap**2 * (sigma**2 + mu**2) / 2


def series_power(sharpe: float, planned: tuple[float, ...], paths: int = 20_000) -> tuple[float, float]:
    """A frozen rule's four-hour returns with annualized Sharpe `sharpe`, Gaussian. The confirmer's wealth is the
    mean of the wealths that bet Kelly's fraction for each planned Sharpe. Returns P(wealth reaches 20) within one
    year and within two."""
    rng = np.random.default_rng(7)
    dt = 1 / STEPS_A_YEAR
    wealth = np.ones((len(planned), paths))
    bets = np.array(planned)[:, None]
    reached = np.zeros(paths, dtype=bool)
    first = reached
    for step in range(2 * STEPS_A_YEAR):
        wealth = wealth * (1 + bets * rng.normal(sharpe * dt, math.sqrt(dt), paths))
        reached |= wealth.mean(axis=0) >= 20
        if step == STEPS_A_YEAR - 1:
            first = reached.copy()
    return float(first.mean()), float(reached.mean())


def confirmer(sigma: float = 0.02) -> None:
    """A frozen return series under three confirmers: the yardstick, the gate's test (bet at most 1/L, L = 2), and
    an exact test whose loss bound is the rule's own, so that Kelly's fraction fits (Argon's finding 2)."""
    annual = math.sqrt(STEPS_A_YEAR)
    print("  Sharpe for 80% power: yardstick; exact test at Kelly's fraction; the gate's capped test")
    for years in (91 / 365, 1, 2):
        steps = round(years * STEPS_A_YEAR)
        spread = 0.5 * sigma * math.sqrt(steps)
        capped = brentq(lambda m: crossing(steps * capped_growth(m, sigma, 0.5), spread) - 0.8, 0, 1)
        print(f"    {steps} steps: {Z80 / math.sqrt(years):.2f}; {U80 / math.sqrt(years):.2f}; {capped * annual / sigma:.1f}")
    for sharpe in (1.4, 2, 2.4):
        print(f"  Kelly's fraction fits under 1/L while the worst step loss L is at most {annual / sharpe:.1f} step standard deviations at Sharpe {sharpe}; the gate's L is {2 / sigma:.0f}")
    print("  power within one year and two, by true Sharpe; betting Kelly's fraction for Sharpe 2 (closed form at two years), and the mean of the bets for 1, 2 and 3:")
    for sharpe in (0, 1, 1.4, 2, 2.4, 3):
        fixed = series_power(sharpe, (2.0,))
        mixed = series_power(sharpe, (1.0, 2.0, 3.0))
        closed = crossing((2 * sharpe - 2) * 2, 2 * math.sqrt(2))
        print(f"    Sharpe {sharpe}: for 2, {fixed[0]:.2f} and {fixed[1]:.2f} ({closed:.2f}); mean of three, {mixed[0]:.2f} and {mixed[1]:.2f}; at Kelly's own fraction {kelly_power(sharpe * math.sqrt(2)) if sharpe else 0:.2f} at two years")
    print(f"  the standard error of a Sharpe measured over two years is {1 / math.sqrt(2):.2f}")
    for sharpe in (1.4, 2.4):
        print(f"  years to 80% power at Sharpe {sharpe}: {U80**2 / sharpe**2:.2f} at Kelly's fraction")


LADDER_NOTIONAL = (1.0, 0.978, 1.864, 3.607, 6.944, 13.24)  # the first buy and five levels, facts-kernel-v2.md section 7
CYCLES_BY_BUYS = (1569, 201, 121, 77, 30, 21)  # cycles with one buy order, two, ..., six or more


def years_to_power(sharpe: float, vol: float, power: float = 0.8) -> float:
    """Years until an account of yearly volatility `vol` and Sharpe `sharpe`, taken as the wealth, has reached 20
    with probability `power`. Infinite when it never does."""
    drift = sharpe * vol - vol**2 / 2

    def shortfall(years: float) -> float:
        return crossing(drift * years, vol * math.sqrt(years)) - power

    return brentq(shortfall, 0.01, 1000) if shortfall(1000) > 0 else math.inf

WINDOWS = (22, 28, 35, 44, 66)


def sign_path(n_max: int, theta1: float, agent_only: float, searcher_only: float) -> tuple[float, float, float, list[str]]:
    """Study 1: Wald's sequential sign test on discordant markets, theta <= 1/2 against theta1 at nominal errors
    5%, ending undecided at n_max markets, when only the agent's pick earns the plant with chance `agent_only` and
    only G100's with `searcher_only`. Exact, by enumeration over (wins, losses). Returns the chance of the claim, of
    the futility stop, the expected markets, and the chance that the claim, and that either boundary, is reached
    within each count of WINDOWS."""
    upper = math.log(0.95 / 0.05)
    lower = math.log(0.05 / 0.95)
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)
    alive = {(0, 0): 1.0}
    up = down = expected = 0.0
    within = []
    for n in range(1, n_max + 1):
        nxt: dict[tuple[int, int], float] = {}
        for (wins, losses), mass in alive.items():
            steps = ((1, 0, agent_only), (0, 1, searcher_only), (0, 0, 1 - agent_only - searcher_only))
            for more_wins, more_losses, chance in steps:
                state = (wins + more_wins, losses + more_losses)
                ratio = state[0] * win + state[1] * loss
                reached = mass * chance
                if ratio >= upper:
                    up += reached
                    expected += n * reached
                elif ratio <= lower:
                    down += reached
                    expected += n * reached
                elif n == n_max:
                    expected += n * reached
                else:
                    nxt[state] = nxt.get(state, 0.0) + reached
        alive = nxt
        if n in WINDOWS:
            within.append(f"{up:.2f}/{up + down:.2f}")
    return up, down, expected, within


def claim_chance(agent: float, searcher: float) -> float:
    """The chance of study 1's claim when the arms earn independently at these rates."""
    return sign_path(100, 0.7, agent * (1 - searcher), searcher * (1 - agent))[0]


def sign_test(n_max: int, theta1: float, cases: tuple[tuple[str, float, float], ...]) -> None:
    """A case is (label, P(only the agent's pick earns the plant), P(only G100's does))."""
    print(f"  theta1 = {theta1}, at most {n_max} markets")
    for label, agent_only, searcher_only in cases:
        up, down, expected, within = sign_path(n_max, theta1, agent_only, searcher_only)
        theta = agent_only / (agent_only + searcher_only)
        print(f"    {label:44s} theta {theta:.3f}: claim {up:.3f}, futility stop {down:.3f}, undecided {1 - up - down:.3f}, markets {expected:.0f}; within {WINDOWS}: {', '.join(within)}")


def pair_rule() -> tuple[int, int]:
    """A pick earns from the plant when it beats half the reference on at least `majority` of its market's `pairs`
    fresh pairs: a binomial test of q, its chance on one pair. The rule is the smallest that classes a pick at
    q = 1/2 with probability at most 0.005 and a pick at q = 3/4 with probability at least 0.9."""
    found = next(
        (pairs, majority)
        for pairs in range(1, 400)
        for majority in range(pairs + 1)
        if binom.sf(majority - 1, pairs, 0.5) <= 0.005 and binom.sf(majority - 1, pairs, 0.75) >= 0.9
    )
    chances = (0.5, 0.6, 0.671, 0.75, 0.8, 0.9)
    for pairs, majority in ((32, 24), found):
        passes = ", ".join(f"q = {q}: {binom.sf(majority - 1, pairs, q):.4f}" for q in chances)
        print(f"  {majority} of {pairs} pairs, the chance a pick is classed as earning: {passes}")
    print(f"  the rule is {found[1]} of {found[0]}; with exactly opposite effects a program and its mirror win disjoint pairs, and 2 * {found[1]} > {found[0]}, so they never both earn")
    return found


def walls(pairs: int, majority: int) -> None:
    """Wall 3 classifies the even part of each of 5 mirror pairs on 200 markets, each pair against its own count.
    Wall 5 holds a program that reads no input below the bottom of G100's band. Argon's momentum: the share of the
    reference's Sharpe that an average of past returns keeps, when the plant is an AR(1) drift."""
    size = binom.sf(majority - 1, pairs, 0.5)
    mirror_pairs, markets = 5, 200

    def any_pair_stops(count: int) -> float:
        return float(1 - (1 - binom.sf(count, markets, size)) ** mirror_pairs)

    allowed = next(count for count in range(markets) if any_pair_stops(count) <= 0.01)
    print(f"  wall 3: an even part at q <= 1/2 earns with probability at most {size:.4f}; more than {allowed} of {markets} for some of the {mirror_pairs} mirror pairs has probability {any_pair_stops(allowed):.4f} (at {allowed - 1}: {any_pair_stops(allowed - 1):.4f}); an even part that earns on 4.5% of markets passes with probability {binom.cdf(allowed, markets, 0.045):.3f}")
    bottom = 0.11
    constant = max(count for count in range(markets) if binom.cdf(count, markets, bottom) <= 0.05)
    grey = ", ".join(f"{rate}: {binom.cdf(constant, markets, rate):.3f}" for rate in (0.03, 0.05, 0.07, 0.11, 0.15))
    print(f"  wall 5: a constant position that earns at the bottom of G100's band, {bottom}, earns on at most {constant} of {markets} markets with probability {binom.cdf(constant, markets, bottom):.3f} (at {constant + 1}: {binom.cdf(constant + 1, markets, bottom):.3f}); the wall passes a constant by its rate, {grey}; one at q = 0.55 on a pair (a rate of {binom.sf(majority - 1, pairs, 0.55):.4f}) with probability {binom.cdf(constant, markets, binom.sf(majority - 1, pairs, 0.55)):.4f}")
    print(f"  an agent that submits one constant on every market, with no search, claims with probability {claim_chance(0.11, 0.11):.3f} when the constant earns at G100's 0.11, and {claim_chance(0.11, 0.19):.3f} against G100's 0.19")
    print(f"  the wall it replaces, 51 of {markets}: a constant that earns at 0.255 passes with probability {binom.cdf(51, markets, 0.255):.3f} and claims with probability {claim_chance(0.255, 0.15):.3f} against G100 at 0.15, {claim_chance(0.255, 0.12):.3f} at 0.12; one at 0.18 passes with {binom.cdf(51, markets, 0.18):.3f} and claims with {claim_chance(0.18, 0.12):.3f} at 0.12 (Argon's round 6 wrote 0.538, 0.768 and 0.284)")
    print(f"  the DSL has one weight expression for all 12 coins: half the reference's effect is {0.5 / math.sqrt(2):.3f} of a six-coin oracle's, at equal variances")
    phi = 1 - 1 / 42
    for snr in (0.03, 0.06, 0.12, 0.25):
        shares = []
        for decay in (0.8, 0.9, 0.95, 0.98):
            covariance = (1 - decay) * phi / (1 - decay * phi) * snr
            variance = (1 - decay) / (1 + decay) * (1 + (1 + decay * phi) / (1 - decay * phi) * snr**2)
            shares.append(f"{covariance / math.sqrt(variance):.2f}")
        print(f"  a momentum average's correlation with the planted drift, at a drift of {snr} step standard deviations (persistence 1 - 1/42), by decay 0.8, 0.9, 0.95, 0.98: {', '.join(shares)}")


def reruns(n_max: int = 100, theta1: float = 0.7) -> None:
    """Why study 1 runs no session again, at the worst: every market discordant at one half, and a rerun that hides a
    loss whenever hiding one raises the chance of the claim. Exact, by backward induction over (wins, losses,
    markets, reruns)."""
    upper = math.log(19)
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)

    @functools.cache
    def claim(wins: int, losses: int, markets: int, left: int) -> float:
        ratio = wins * win + losses * loss
        if ratio >= upper:
            return 1.0
        if ratio <= -upper or markets == n_max:
            return 0.0
        kept = claim(wins, losses + 1, markets + 1, left)
        hidden = claim(wins, losses, markets, left - 1) if left else 0.0
        return 0.5 * claim(wins + 1, losses, markets + 1, left) + 0.5 * max(kept, hidden)

    sizes = ", ".join(f"{count}: {claim(0, 0, 0, count):.3f}" for count in range(6))
    print(f"  the size when up to v reruns each hide a loss, by v: {sizes}")
    for share in (19 / 40, 89 / 128):
        print(f"  were a session with a fallback run again (a share of {share:.3f}), a counted market would cost {1 / (1 - share):.1f} sessions: {29 / (1 - share):.0f} for 29 markets, {41 / (1 - share):.0f} for 41")


def calibration(pairs: int) -> None:
    """The strength is where G100's rate on `search` markets is within 0.005 of 0.15; it is accepted when the rate
    on `check` fresh markets is within `tolerance` of 0.15. A false stop is that check failing by sampling alone."""
    target = 0.15 * 0.85

    def false_stop(search: int, check: int, tolerance: float, offset: float) -> float:
        spread = math.sqrt(target / search + target / check)
        return float(norm.sf((tolerance - offset) / spread) + norm.sf((tolerance + offset) / spread))

    print(f"  revision 4, a search on 400 markets and a check on 2,000 within 0.02: a false stop with probability {false_stop(400, 2000, 0.02, 0):.2f}")
    search, check, tolerance = 1600, 10_000, 0.03
    edge = math.sqrt(0.19 * 0.81 / check)
    print(f"  a search on {search:,} and a check on {check:,} within {tolerance}: a false stop with probability {false_stop(search, check, tolerance, 0.005):.4f}; an accepted strength has a true rate within {tolerance + 2.5 * edge:.3f} of 0.15 (2.5 standard errors of {edge:.4f} at 0.19)")
    market = 100 + 2 * 2 * pairs
    steps = (9, 13)
    wall3 = 200 * 2 * pairs * (10 + 1)
    print(f"  kernel evaluations: {3 * 2 * pairs} a study market to classify both picks and the reference, {100 * 3 * 2 * pairs:,} for 100; a calibration market at one strength, {market}; the search at {steps[0]} strengths (at most {steps[1]}) and the check: {(steps[0] * search + check) * market:,} (at most {(steps[1] * search + check) * market:,}); wall 3: {wall3:,}")


def study1() -> None:

        rows.append((f"0.5 with {lost} of sessions lost, 0.19, least", agent, 0.19))

    rows.append(("0.5, 0.3, least overlap (Argon)", 0.5, 0.3))
    rows.append(("0.25, 0.2, both 0.1875 (Argon)", 0.0625, 0.0125))

def expected_max(count: int) -> float:
    """The expected maximum of `count` independent standard Gaussians, by quadrature."""
    value, _ = quad(lambda x: x * count * norm.pdf(x) * norm.cdf(x) ** (count - 1), -10, 12, limit=200)
    return float(value)


def dsr_z(sharpe: float, threshold: float, steps: int, skew: float, kurtosis: float) -> float:
    spread = math.sqrt(1 - skew * sharpe + (kurtosis - 1) / 4 * sharpe**2)
    return (sharpe - threshold) * math.sqrt(steps - 1) / spread
