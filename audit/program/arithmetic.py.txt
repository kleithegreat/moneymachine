import json

import numpy as np
from scipy.integrate import quad
from scipy.optimize import brentq
from scipy.stats import beta, binom, norm

Z80 = norm.ppf(0.95) + norm.ppf(0.80)
LOG20 = math.log(20)
STEPS_A_YEAR = 6 * 365
STAKE_CAP = 0.05
SESSIONS_A_WEEK = 22  # an assumption: TAMU and Cornell together


def yardstick() -> None:
    """The Gaussian fixed-horizon yardstick, z = S * sqrt(T): a planning reference, not a limit on every test."""
    print(f"z* = {Z80:.3f}, z*^2 = {Z80**2:.2f}")
    for sharpe in (1, 1.4, 2, 2.4, 5):
        print(f"  Sharpe {sharpe}: {Z80**2 / sharpe**2:.2f} years at 80% power")
    for years in (91 / 365, 1, 2):
        print(f"  T = {years:.3f} years: Sharpe {Z80 / math.sqrt(years):.2f} at 80% power")


def crossing(drift: float, spread: float) -> float:
    """P(a Brownian log wealth with total drift `drift` and total standard deviation `spread` reaches log 20
    at any time before the end): the reflection formula. The test stops at 20, so this is its power."""
    reflected = math.exp(min(0.0, 2 * drift * LOG20 / spread**2 + norm.logcdf((-drift - LOG20) / spread)))
    return float(norm.cdf((drift - LOG20) / spread)) + reflected


def kelly_power(u: float) -> float:
    """Power of an exact test staked at Kelly's fraction, where u = effect * sqrt(count)."""
    return crossing(u * u / 2, u)


U80 = brentq(lambda u: kelly_power(u) - 0.8, 1, 6)


def exact_cost() -> None:
    """Independent even bets with net edge e, staked at Kelly's fraction e: log wealth after N bets is near
    Gaussian with mean N*e^2/2 and standard deviation e*sqrt(N)."""
    z = norm.ppf(0.80)
    terminal = z + math.sqrt(z * z + 2 * LOG20)
    print(f"  stopping at 20: N = {U80**2:.2f} / e^2 for 80% power (u = {U80:.3f}; Argon wrote 8.67); the yardstick: {Z80**2:.2f} / e^2, so the exact test costs {U80**2 / Z80**2:.2f} times as much")
    print(f"  revision 3 asked for wealth of 20 at the end: N = {terminal**2:.1f} / e^2, where the power is really {kelly_power(terminal):.3f}")
    for edge in (0.03, 0.05, 0.08, 0.1):
        print(f"    net edge {edge}: {U80**2 / edge**2:,.0f} independent bets")
    print(f"  within 104 weeks: count a week * e^2 >= {U80**2 / 104:.4f}")
    for bets in (5, 10, 15, 33, 45, 60):
        edge = U80 / math.sqrt(104 * bets)
        print(f"    {bets} bets a week: net edge {edge:.3f} (a hit rate of {(1 + edge) / 2:.3f} on an even bet); power at 0.03, 0.05, 0.08: " + ", ".join(f"{kelly_power(e * math.sqrt(104 * bets)):.2f}" for e in (0.03, 0.05, 0.08)))
    for edge in (0.03, 0.05, 0.08):
        print(f"    net edge {edge}: {U80**2 / (104 * edge**2):.0f} bets a week")
    print(f"  discordant paired forecasts for a hit-rate gain of 0.05 (edge 0.1): {U80**2 / 0.1**2:.0f}; of 0.025: {U80**2 / 0.05**2:.0f}")


def account(bets: int, hold: int, edge: float, fraction: float, weeks: int = 104, paths: int = 4000) -> float:
    """A paper account in even bets with net edge `edge`: `bets` purchases a week, each staking `fraction` of the
    cash left at that moment and paid `hold` purchases later (0: before the next purchase). Returns P(cash reaches
    20 within `weeks`)."""
    rng = np.random.default_rng(11)
    cash = np.ones(paths)
    pending = np.zeros((paths, hold + 1))
    reached = np.zeros(paths, dtype=bool)
    for bet in range(weeks * bets):
        slot = bet % (hold + 1)
        cash = cash + pending[:, slot]
        reached |= cash >= 20
        stake = fraction * cash
        cash = cash - stake
        pending[:, slot] = 2 * stake * (rng.random(paths) < (1 + edge) / 2)
    reached |= cash + pending.sum(axis=1) >= 20
    return float(reached.mean())


def forward_account() -> None:
    """The account's sequential rule simulated with three settlement timings: before the next purchase, a day
    later, and a week later (revision 3's batch, which Argon's finding 2 showed starves the stake)."""
    print(f"  no edge, 33 bets a week staking the cap, paid a day later: {account(33, 5, 0.0, STAKE_CAP, paths=20_000):.3f} of paths reach 20 (Ville's bound: 0.05 over any horizon)")
    print("  net edge for 80% power within 104 weeks, stake min(5%, Kelly) of cash:")
    for bets in (5, 10, 15, 33, 60):
        found = []
        for hold in (0, math.ceil(bets / 7), bets):
            found.append(brentq(lambda e: account(bets, hold, e, min(STAKE_CAP, e)) - 0.8, 0.01, 0.6, xtol=0.001))
        print(f"    {bets:2d} bets a week: paid at once {found[0]:.3f}, a day later {found[1]:.3f} (a hit rate of {(1 + found[1]) / 2:.3f}), a week later {found[2]:.3f}; closed form {U80 / math.sqrt(104 * bets):.3f}")
    for bets in (10, 33):
        hold = math.ceil(bets / 7)
        print(f"    {bets} bets a week paid a day later, power at 0.03, 0.05, 0.08: " + ", ".join(f"{account(bets, hold, e, min(STAKE_CAP, e)):.2f}" for e in (0.03, 0.05, 0.08)))


def capped_growth(mu: float, sigma: float, cap: float) -> float:
    """Expected log growth a step of wealth bet at the constant fraction `cap`, to second order."""
    return cap * mu - cap**2 * (sigma**2 + mu**2) / 2


def series_power(sharpe: float, planned: tuple[float, ...], paths: int = 20_000) -> tuple[float, float]:
    """A frozen rule's four-hour returns with annualized Sharpe `sharpe`, Gaussian. The confirmer's wealth is the
    mean of the wealths that bet Kelly's fraction for each planned Sharpe. Returns P(wealth reaches 20) within one
    year and within two."""
    rng = np.random.default_rng(7)
    dt = 1 / STEPS_A_YEAR
    wealth = np.ones((len(planned), paths))
    bets = np.array(planned)[:, None]
    reached = np.zeros(paths, dtype=bool)
    first = reached
    for step in range(2 * STEPS_A_YEAR):
        wealth = wealth * (1 + bets * rng.normal(sharpe * dt, math.sqrt(dt), paths))
        reached |= wealth.mean(axis=0) >= 20
        if step == STEPS_A_YEAR - 1:
            first = reached.copy()
    return float(first.mean()), float(reached.mean())


def confirmer(sigma: float = 0.02) -> None:
    """A frozen return series under three confirmers: the yardstick, the gate's test (bet at most 1/L, L = 2), and
    an exact test whose loss bound is the rule's own, so that Kelly's fraction fits (Argon's finding 2)."""
    annual = math.sqrt(STEPS_A_YEAR)
    print("  Sharpe for 80% power: yardstick; exact test at Kelly's fraction; the gate's capped test")
    for years in (91 / 365, 1, 2):
        steps = round(years * STEPS_A_YEAR)
        spread = 0.5 * sigma * math.sqrt(steps)
        capped = brentq(lambda m: crossing(steps * capped_growth(m, sigma, 0.5), spread) - 0.8, 0, 1)
        print(f"    {steps} steps: {Z80 / math.sqrt(years):.2f}; {U80 / math.sqrt(years):.2f}; {capped * annual / sigma:.1f}")
    for sharpe in (1.4, 2, 2.4):
        print(f"  Kelly's fraction fits under 1/L while the worst step loss L is at most {annual / sharpe:.1f} step standard deviations at Sharpe {sharpe}; the gate's L is {2 / sigma:.0f}")
    print("  power within one year and two, by true Sharpe; betting Kelly's fraction for Sharpe 2 (closed form at two years), and the mean of the bets for 1, 2 and 3:")
    for sharpe in (0, 1, 1.4, 2, 2.4, 3):
        fixed = series_power(sharpe, (2.0,))
        mixed = series_power(sharpe, (1.0, 2.0, 3.0))
        closed = crossing((2 * sharpe - 2) * 2, 2 * math.sqrt(2))
        print(f"    Sharpe {sharpe}: for 2, {fixed[0]:.2f} and {fixed[1]:.2f} ({closed:.2f}); mean of three, {mixed[0]:.2f} and {mixed[1]:.2f}; at Kelly's own fraction {kelly_power(sharpe * math.sqrt(2)) if sharpe else 0:.2f} at two years")
    print(f"  the standard error of a Sharpe measured over two years is {1 / math.sqrt(2):.2f}")
    for sharpe in (1.4, 2.4):
        print(f"  years to 80% power at Sharpe {sharpe}: {U80**2 / sharpe**2:.2f} at Kelly's fraction")


def sign_test(n_max: int, theta1: float, cases: tuple[tuple[str, float, float], ...]) -> None:
    """Study 1: Wald's sequential sign test on discordant markets, theta <= 1/2 against theta1 at nominal errors
    5%, ending undecided at n_max markets. A case is (label, P(only the agent's pick earns the plant),
    P(only G100's does)). Exact, by enumeration over (wins, losses)."""
    upper = math.log(0.95 / 0.05)
    lower = math.log(0.05 / 0.95)
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)
    print(f"  theta1 = {theta1}, at most {n_max} markets")
    for label, agent_only, searcher_only in cases:
        alive = {(0, 0): 1.0}
        up = down = expected = 0.0
        for n in range(1, n_max + 1):
            nxt: dict[tuple[int, int], float] = {}
            for (wins, losses), mass in alive.items():
                steps = ((1, 0, agent_only), (0, 1, searcher_only), (0, 0, 1 - agent_only - searcher_only))
                for more_wins, more_losses, chance in steps:
                    state = (wins + more_wins, losses + more_losses)
                    ratio = state[0] * win + state[1] * loss
                    reached = mass * chance
                    if ratio >= upper:
                        up += reached
                        expected += n * reached
                    elif ratio <= lower:
                        down += reached
                        expected += n * reached
                    elif n == n_max:
                        expected += n * reached
                    else:
                        nxt[state] = nxt.get(state, 0.0) + reached
            alive = nxt
        theta = agent_only / (agent_only + searcher_only)
        print(f"    {label:44s} theta {theta:.3f}: claim {up:.3f}, futility stop {down:.3f}, undecided {1 - up - down:.3f}, markets {expected:.0f}")


def study1() -> None:
    """The joint law of the two arms on one market is (agent's rate, G100's rate, both). The size holds for every
    joint law with the agent's rate at most G100's; power depends on all three, so every row is printed."""
    rows: list[tuple[str, float, float]] = [("null, every market discordant", 0.5, 0.5)]
    for rate in (0.11, 0.15, 0.19):
        rows.append((f"null {rate}, {rate}, independent", rate * (1 - rate), rate * (1 - rate)))
    for agent in (0.5, 0.4, 0.3):
        for searcher in (0.11, 0.15, 0.19):
            for name, both in (("least", 0.0), ("independent", agent * searcher), ("most", searcher)):
                rows.append((f"{agent}, {searcher}, {name} overlap", agent - both, searcher - both))
    rows.append(("0.5, 0.25, least overlap (revision 3's band edge)", 0.5, 0.25))
    rows.append(("0.5, 0.3, least overlap (Argon)", 0.5, 0.3))
    rows.append(("0.25, 0.2, both 0.1875 (Argon)", 0.0625, 0.0125))
    sign_test(100, 0.7, tuple(rows))
    print(f"  a discordant win adds {math.log(0.7 / 0.5):.4f}, a loss {math.log(0.3 / 0.5):.4f}; the boundaries are +-{math.log(19):.4f}: {math.ceil(math.log(19) / math.log(1.4))} straight wins claim, {math.ceil(math.log(19) / -math.log(0.6))} straight losses stop")
    for markets in (400, 2000):
        print(f"  G100's rate near 0.15 on {markets} calibration markets has standard error {math.sqrt(0.15 * 0.85 / markets):.4f}")
    print(f"  G100 at 0.3 is {(0.3 - 0.17) / math.sqrt(0.15 * 0.85 / 2000):.0f} standard errors above the band's 0.17; rates 0.25 and 0.2 with 0.1875 in common leave {100 * (0.0625 + 0.0125):.1f} discordant markets in 100")
    print("  a pick earns from the plant when it beats half the reference on at least 24 of its market's 32 fresh pairs; by its chance q on one pair:")
    print("    " + ", ".join(f"q = {q}: {binom.sf(23, 32, q):.4f}" for q in (0.5, 0.6, 0.75, 0.9, 0.95)))
    print(f"  kernel evaluations: {3 * 2 * 32} a study market to classify both picks and the reference, {100 * 3 * 2 * 32:,} for 100; calibration, 8 strengths of 400 markets and one of 2,000: {(8 * 400 + 2000) * (100 + 2 * 2 * 32):,}")
    spent = [json.loads(line)["agent_run"] for line in Path("audit/stage3/null-v3.jsonl").read_text().splitlines()]
    counts = sorted(run["spent"] for run in spent)
    print(f"  stage 3's {len(spent)} null campaigns: the agent spent {counts[0]} to {counts[-1]} of 100 evaluations, median {statistics.median(counts)}; stopped: {sorted({run['stopped'] for run in spent})}")
    print(f"  at {SESSIONS_A_WEEK} sessions a week (an assumption), 100 markets take {100 / SESSIONS_A_WEEK:.1f} weeks, 41 take {41 / SESSIONS_A_WEEK:.1f}")


def expected_max(count: int) -> float:
    """The expected maximum of `count` independent standard Gaussians, by quadrature."""
    value, _ = quad(lambda x: x * count * norm.pdf(x) * norm.cdf(x) ** (count - 1), -10, 12, limit=200)
    return float(value)


def dsr_z(sharpe: float, threshold: float, steps: int, skew: float, kurtosis: float) -> float:
    spread = math.sqrt(1 - skew * sharpe + (kurtosis - 1) / 4 * sharpe**2)
    return (sharpe - threshold) * math.sqrt(steps - 1) / spread


def argon_paper() -> None:
    """Rechecks of Argon's findings on the paper (audit/program/argon-paper-findings.md)."""
    print("  4. Clopper-Pearson upper bounds at 95%, no promotion in n:")
    for n in (9, 10, 13, 21):
        print(f"       n = {n}: {1 - 0.05 ** (1 / n):.3f}")
    rho = 8 / math.sqrt(46.25 * 2.6)
    print(f"  5. correlation {rho:.4f}; a null survivor at z = 3 passes the sign check with probability {norm.cdf(3 * rho / math.sqrt(1 - rho**2)):.4f}")
    print(f"  6. Hoeffding cutoff {0.2 - math.sqrt(math.log(20) / 200):.4f}")
    year = math.sqrt(STEPS_A_YEAR)
    for steps in (100, 546, 5400):
        print(f"  7. T = {steps}: the limit of the DSR's z as the Sharpe grows is under 1.645 only above kurtosis {1 + 4 * (steps - 1) / norm.ppf(0.95) ** 2:.0f}; a sample's is at most {steps}")
    print(f"     Argon's examples at T = 546 against a threshold of 2: Sharpe 3.5 has z at most {dsr_z(3.5 / year, 2 / year, 546, 0, 1):.3f}; Sharpe 6, skewness -4, kurtosis 18 has z = {dsr_z(6 / year, 2 / year, 546, -4, 18):.3f}; the limit at kurtosis 18 is {2 * math.sqrt(545 / 17):.1f}")
    worst = min((1 - skew * sharpe / 2) ** 2 for skew in np.linspace(-6, 6, 25) for sharpe in np.linspace(0, 3, 31))
    print(f"     a sample's kurtosis is at least its skewness squared plus 1, so the DSR's radicand is at least (1 - skewness * Sharpe / 2)^2 >= {worst:.0f}; Argon's skewness 4 with kurtosis 10 is no sample's ({4**2 + 1} at least)")
    for level, tolerance, inflated in ((0.05, 0.0625, 0.056), (0.01, 0.015, 0.012), (0.005, 0.01, 0.0075)):
        suite = max(x for x in range(2000) if beta.ppf(1 - 0.05 / 99, x + 1, 20_000 - x) <= tolerance)
        single = max(x for x in range(2000) if beta.ppf(0.95, x + 1, 20_000 - x) <= tolerance)
        chance = binom.cdf(suite, 20_000, inflated)
        print(f"  10. level {level}, tolerance {tolerance}: up to {suite} rejections in 20,000 pass at the suite's confidence 1 - 0.05/99 ({single} at a single cell's 95%); a true rate of {inflated} passes with probability {chance:.2f}")
    phi, keep = 1 - 1 / 42, 5 / 6
    whole, kept = (1 + phi) / (1 - phi), (1 + phi * keep) / (1 - phi * keep)
    print(f"  11. a funding leg's long-run variance factor is {whole:.1f}, of which blocks of mean 6 keep {kept:.2f}")
    for share in (0.000025, 0.001, 0.01, 0.05, 0.1, 0.25):
        print(f"      funding at {share:g} of a step's variance: the bootstrap understates the variance of the mean by {(1 - share + share * whole) / (1 - share + share * kept):.3f}")
    share = brentq(lambda w: (1 - w + w * whole) / (1 - w + w * kept) - 1.1, 0, 1)
    print(f"      the understatement reaches 10% at a share of {share:.4f}: a funding standard deviation of {100 * math.sqrt(share) * 0.02:.3f}% a step beside 2% of price")
    print("  13. expected maximum of N Gaussians against sqrt(2 log N), and the ceiling at 3, 10, 20 years:")
    for count in (10, 1000, 100_000):
        exact = expected_max(count)
        ceilings = ", ".join(f"{exact / math.sqrt(years):.2f}" for years in (3, 10, 20))
        print(f"       N = {count}: {exact:.4f} against {math.sqrt(2 * math.log(count)):.4f}; {ceilings}")
    se = math.sqrt(365 / 550)
    print(f"  14. standard error {se:.2f}; a Sharpe 2 leak stays under the canary's alarm at 4 with probability {norm.cdf((4 - 2) / se):.3f}")
