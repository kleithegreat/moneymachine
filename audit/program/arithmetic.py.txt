"""Reproduces the numbers in audit/program/architecture.md (revision 2), audit/program/paper-changes.md and

import math

import numpy as np
from scipy.integrate import quad
from scipy.optimize import brentq
from scipy.stats import beta, binom, norm

Z80 = norm.ppf(0.95) + norm.ppf(0.80)
LOG20 = math.log(20)
STEPS_A_YEAR = 6 * 365


def yardstick() -> None:
    """The Gaussian fixed-horizon yardstick, z = S * sqrt(T): a planning reference, not a limit on every test."""
    print(f"z* = {Z80:.3f}, z*^2 = {Z80**2:.2f}; Holm over 3: {norm.ppf(1 - 0.05 / 3) + norm.ppf(0.8):.2f}")
    for sharpe in (0.5, 1, 1.5, 2, 3, 5):
        years80 = Z80**2 / sharpe**2
        years50 = norm.ppf(0.95) ** 2 / sharpe**2
        print(f"  Sharpe {sharpe}: {years80:.2f} years at 80% power ({12 * years80:.1f} months), {years50:.2f} at even odds")
    for years in (60 / 365, 91 / 365, 1, 1.3, 1.8, 2.2):
        print(f"  T = {years:.3f} years: Sharpe {Z80 / math.sqrt(years):.2f} at 80% power")


def capped_bet(sigma: float = 0.02, cap: float = 0.5) -> None:
    """The exact test: wealth must reach 20. Expected log growth a step at a constant bet lam is
    lam*mu - lam^2*(sigma^2 + mu^2)/2, and lam may not exceed 1/L (the gate: L = 2)."""
    for steps in (360, 546, 1080, STEPS_A_YEAR, 2 * STEPS_A_YEAR, 5400):
        free = math.sqrt(2 * LOG20 * STEPS_A_YEAR / steps)
        kelly = free / math.sqrt(STEPS_A_YEAR) / sigma
        mu = brentq(lambda m: steps * (cap * m - cap**2 * (sigma**2 + m**2) / 2) - LOG20, 0, 1)
        capped = mu / sigma * math.sqrt(STEPS_A_YEAR)
        print(f"  {steps} steps: Kelly's bet {kelly:.2f} would need Sharpe {free:.2f}; capped at {cap}: mean {100 * mu:.2f}% a step, Sharpe {capped:.1f}")
    mu = 4.9 / math.sqrt(STEPS_A_YEAR) * sigma
    growth = 546 * (cap * mu - cap**2 * (sigma**2 + mu**2) / 2)
    print(f"  Sharpe 4.9 at the cap over 546 steps: expected log wealth {growth:.3f} against {LOG20:.3f}")


def rates() -> None:
    """Months to 80% power on the yardstick, 6.18 / (d^2 * n)."""
    rows = (("theses on perps", 0.05, 15), ("theses on perps", 0.05, 40), ("theses on perps", 0.1, 15),
            ("theses on perps", 0.1, 40), ("events on returns", 0.1, 20), ("events on returns", 0.1, 40),
            ("events on returns", 0.2, 20), ("events on returns", 0.2, 40))  # fmt: skip
    for name, d, n in rows:
        print(f"  {name:18s} d = {d:<4} n = {n:3d} a month: {Z80**2 / (d * d * n):6.1f} months")
    for ic in (0.03, 0.05):
        sharpe = ic * math.sqrt(52 * 20)
        print(f"  breadth, IC {ic}, 20 bets a week: Sharpe {sharpe:.2f}, {12 * Z80**2 / sharpe**2:.0f} months")
    for q in (0.5, 0.01):
        print(f"  paired Brier, main forecast exact, |e| = 0.05, q = {q}: effect {0.05 / (2 * math.sqrt(q * (1 - q))):.4f}")


def sign_test(n_max: int, theta1: float = 0.8, alpha: float = 0.05, beta: float = 0.05) -> None:
    """Study 1: Wald's sequential sign test on discordant markets (exactly one arm's pick carries the plant),
    theta <= 1/2 against theta1, ending undecided at n_max markets. Exact, by enumeration over (wins, losses)."""
    upper = math.log((1 - beta) / alpha)
    lower = math.log(beta / (1 - alpha))
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)
    print(f"  at most {n_max} markets; a win adds {win:.3f}, a loss {loss:.3f}; boundaries {upper:.3f} and {lower:.3f}")
    arms = ((0.2, 0.2), (0.5, 0.5), (0.3, 0.2), (0.35, 0.2), (0.5, 0.2), (0.6, 0.2), (0.8, 0.3), (0.5, 0.1), (0.5, 0.3))
    cases = [("every market discordant, even", 0.5, 0.5)]
    for agent, searcher in arms:
        cases.append((f"agent {agent}, G100 {searcher}", agent * (1 - searcher), searcher * (1 - agent)))
    for label, agent_only, searcher_only in cases:
        alive = {(0, 0): 1.0}
        up = down = expected = 0.0
        for n in range(1, n_max + 1):
            nxt: dict[tuple[int, int], float] = {}
            for (wins, losses), mass in alive.items():
                steps = ((1, 0, agent_only), (0, 1, searcher_only), (0, 0, 1 - agent_only - searcher_only))
                for more_wins, more_losses, chance in steps:
                    state = (wins + more_wins, losses + more_losses)
                    ratio = state[0] * win + state[1] * loss
                    reached = mass * chance
                    if ratio >= upper:
                        up += reached
                        expected += n * reached
                    elif ratio <= lower:
                        down += reached
                        expected += n * reached
                    elif n == n_max:
                        expected += n * reached
                    else:
                        nxt[state] = nxt.get(state, 0.0) + reached
            alive = nxt
        theta = agent_only / (agent_only + searcher_only)
        print(f"    {label} (theta {theta:.3f}): upper {up:.4f}, lower {down:.4f}, undecided {1 - up - down:.4f}, expected markets {expected:.1f}")


def student_fails() -> None:
    """GPT's null: 60 events a week at price 0.50; with probability 30/31 exactly 31 pay, otherwise none do."""
    for weeks in (26, 52):
        print(f"  no losing week in {weeks}: {(30 / 31) ** weeks:.4f} (the Student statistic is then infinite)")
    weeks = math.ceil(LOG20 / math.log(31 / 30))
    print(f"  the same null against the account: {weeks} winning weeks to reach 20, probability {(30 / 31) ** weeks:.4f}")


def account(cost: float, edge: float, bets: int, deployed: float, rng: np.random.Generator) -> str:
    """Study 2's power, for planning only: `bets` independent contracts a week at all-in cost `cost` and true
    probability cost*(1 + edge), the share `deployed` of cash staked each week in equal parts, all settled
    within the week. Validity needs none of this; power does."""
    paths, weeks = 20_000, 156
    wins = rng.binomial(bets, cost * (1 + edge), size=(paths, weeks))
    factor = 1 - deployed + deployed * wins / (bets * cost)
    peak = np.maximum.accumulate(np.cumprod(factor, axis=1), axis=1)
    growth = float(np.mean(np.log(np.maximum(factor, 1e-300))))
    reach = ", ".join(f"{(peak[:, w - 1] >= 20).mean():.2f}" for w in (26, 52, 104, 156))
    return f"growth {growth:.4f} a week, {LOG20 / growth:5.0f} weeks expected; {reach}"


def study2() -> None:
    rng = np.random.default_rng(11)
    print("  edge is expected net return per dollar staked; columns: P(cash reaches 20 by week 26, 52, 104, 156)")
    for deployed in (1.0, 0.5):
        for bets in (30, 60):
            for edge in (0.02, 0.04, 0.08, 0.12, 0.16):
                print(f"    cost 0.5, deployed {deployed}, {bets} bets a week, edge {edge:.2f}: {account(0.5, edge, bets, deployed, rng)}")
    print("  the same probability edge at other prices (60 bets a week, fully deployed)")
    for cost in (0.2, 0.5, 0.8):
        for points in (0.02, 0.04):
            print(f"    cost {cost}, {points:.2f} of probability (edge {points / cost:.3f}): {account(cost, points / cost, 60, 1.0, rng)}")

    print(f"  twenty null instances at 5%: at least one rejection with probability {1 - 0.95**20:.3f}")
    for bets, edge in ((60, 0.04), (60, 0.08), (30, 0.08)):
        print(f"  {bets} even bets, edge {edge}: log growth a turnover about {edge - 1 / (2 * bets) - edge**2 / 2:.4f}")


def expected_max(count: int) -> float:
    """The expected maximum of `count` independent standard Gaussians, by quadrature."""
    value, _ = quad(lambda x: x * count * norm.pdf(x) * norm.cdf(x) ** (count - 1), -10, 12, limit=200)
    return float(value)


def argon() -> None:
    """Rechecks of Argon's findings on the paper (audit/program/argon-paper-findings.md)."""
    print("  4. Clopper-Pearson upper bounds at 95%, no promotion in n:")
    for n in (9, 10, 13, 21):
        print(f"       n = {n}: {1 - 0.05 ** (1 / n):.3f}")
    print(f"     one promotion in 22: {beta.ppf(0.95, 2, 21):.3f}; in 14: {beta.ppf(0.95, 2, 13):.3f}")
    rho = 8 / math.sqrt(46.25 * 2.6)
    print(f"  5. correlation {rho:.4f}; a null survivor at z = 3 passes the sign check with probability {norm.cdf(3 * rho / math.sqrt(1 - rho**2)):.4f}")
    print(f"  6. Hoeffding cutoff {0.2 - math.sqrt(math.log(20) / 200):.4f}; over 9 cells {0.2 - math.sqrt(math.log(180) / 200):.4f}")
    for steps in (100, 546, 5400):
        most = steps - 2 + 1 / (steps - 1)
        print(f"  7. T = {steps}: the DSR ceiling is under 0.95 only above kurtosis {1 + 4 * (steps - 1) / norm.ppf(0.95) ** 2:.0f}; a sample's is at most {most:.0f}")
    for level, tolerance, inflated in ((0.05, 0.0625, 0.056), (0.01, 0.015, 0.012), (0.005, 0.01, 0.0075)):
        passing = max(x for x in range(2000) if beta.ppf(1 - 0.05 / 99, x + 1, 20_000 - x) <= tolerance)
        chance = binom.cdf(passing, 20_000, inflated)
        print(f"  10. level {level}, tolerance {tolerance}: up to {passing} rejections in 20,000 pass; a true rate of {inflated} passes with probability {chance:.2f}")
    phi, keep = 1 - 1 / 42, 5 / 6
    print(f"  11. long-run variance factor {(1 + phi) / (1 - phi):.1f}, of which blocks of mean 6 keep {(1 + phi * keep) / (1 - phi * keep):.2f}")
    print("  13. expected maximum of N Gaussians against sqrt(2 log N), and the ceiling at 3, 10, 20 years:")
    for count in (10, 1000, 100_000):
        exact = expected_max(count)
        ceilings = ", ".join(f"{exact / math.sqrt(years):.2f}" for years in (3, 10, 20))
        print(f"       N = {count}: {exact:.4f} against {math.sqrt(2 * math.log(count)):.4f}; {ceilings}")
    se = math.sqrt(365 / 550)
    print(f"  14. a Sharpe 2 leak stays under the canary's alarm at 4 with probability {norm.cdf((4 - 2) / se):.3f}")
    for runs in (1, 20, 220):
        print(f"      mean of {runs} runs: standard error {se / math.sqrt(runs):.3f} of Sharpe")
