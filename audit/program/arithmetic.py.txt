import math

import numpy as np
from scipy.integrate import quad
from scipy.optimize import brentq
from scipy.stats import beta, binom, norm

Z80 = norm.ppf(0.95) + norm.ppf(0.80)
LOG20 = math.log(20)
STEPS_A_YEAR = 6 * 365
STAKE_CAP = 0.05


def yardstick() -> None:
    """The Gaussian fixed-horizon yardstick, z = S * sqrt(T): a planning reference, not a limit on every test."""
    print(f"z* = {Z80:.3f}, z*^2 = {Z80**2:.2f}")
    for sharpe in (0.5, 1, 1.5, 2, 3, 5):
        years = Z80**2 / sharpe**2
        print(f"  Sharpe {sharpe}: {years:.2f} years at 80% power ({12 * years:.1f} months)")
    for years in (91 / 365, 1, 2):
        print(f"  T = {years:.3f} years: Sharpe {Z80 / math.sqrt(years):.2f} at 80% power")


def capped_growth(mu: float, sigma: float, cap: float) -> float:
    """Expected log growth a step of wealth bet at the constant fraction `cap`, to second order."""
    return cap * mu - cap**2 * (sigma**2 + mu**2) / 2


def capped_bet(sigma: float = 0.02, cap: float = 0.5) -> None:
    """The gate's exact test: wealth must reach 20 and the bet may not exceed 1/L (L = 2). Two thresholds a
    length: expected log wealth equal to log 20 (about even odds), and 80% power, taking log wealth as Gaussian
    with standard deviation cap * sigma * sqrt(steps) (Argon's round 2, finding 7)."""
    for steps in (360, 546, 1080, 2 * STEPS_A_YEAR, 5400):
        spread = norm.ppf(0.80) * cap * sigma * math.sqrt(steps)
        even = brentq(lambda m: steps * capped_growth(m, sigma, cap) - LOG20, 0, 1)
        power = brentq(lambda m: steps * capped_growth(m, sigma, cap) - LOG20 - spread, 0, 1)
        annual = math.sqrt(STEPS_A_YEAR) / sigma
        print(f"  {steps} steps: Sharpe {even * annual:.1f} at even odds, {power * annual:.1f} at 80% power")
    mu = 4.9 / math.sqrt(STEPS_A_YEAR) * sigma
    print(f"  Sharpe 4.9: Kelly's bet {mu / sigma**2:.1f} against the cap {cap}; expected log wealth over 546 steps {546 * capped_growth(mu, sigma, cap):.2f} against {LOG20:.2f}")


def betting_cost() -> None:
    """Independent even bets with net edge e per unit of worst loss, staked at Kelly's fraction e: log wealth
    after N bets is near Gaussian with mean N*e^2/2 and standard deviation e*sqrt(N). With u = e*sqrt(N), 80%
    power needs u^2/2 - z80*u = log 20."""
    z = norm.ppf(0.80)
    u = z + math.sqrt(z * z + 2 * LOG20)
    print(f"  an exact test at Kelly's stake: N = {u * u:.1f} / e^2 for 80% power, {2 * LOG20:.1f} / e^2 at even odds; the yardstick: {Z80**2:.2f} / e^2")
    for edge in (0.025, 0.05, 0.08, 0.1):
        print(f"    net edge {edge}: {u * u / edge**2:,.0f} independent bets")
    for bets in (5, 10, 15, 30, 60):
        print(f"    {bets} bets a week for 104 weeks: 80% power at a net edge of {u / math.sqrt(104 * bets):.3f} if the stake cap did not bind")

    rows = (("theses on perps", 0.05, 15), ("theses on perps", 0.05, 40), ("theses on perps", 0.1, 15),
            ("theses on perps", 0.1, 40))  # fmt: skip
    for name, d, n in rows:
        print(f"  {name} d = {d:<4} n = {n:3d} a month: {Z80**2 / (d * d * n):6.1f} months")

        sharpe = ic * math.sqrt(52 * 20)
        print(f"  breadth, IC {ic}, 20 bets a week: Sharpe {sharpe:.2f}, {12 * Z80**2 / sharpe**2:.0f} months")
    print(f"  a 0.03 Brier gain over a forecast of 0.50 needs a true probability of {0.5 + math.sqrt(0.03):.3f}")


def sign_test(n_max: int, theta1: float, cases: tuple[tuple[str, float, float], ...]) -> None:
    """Study 1: Wald's sequential sign test on discordant markets, theta <= 1/2 against theta1 at nominal errors
    5%, ending undecided at n_max markets. A case is (label, P(only the agent's pick earns the plant),
    P(only G100's does)). Exact, by enumeration over (wins, losses)."""
    upper = math.log(0.95 / 0.05)
    lower = math.log(0.05 / 0.95)
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)
    print(f"  theta1 = {theta1}, at most {n_max} markets")
    for label, agent_only, searcher_only in cases:
        alive = {(0, 0): 1.0}
        up = down = expected = 0.0
        for n in range(1, n_max + 1):
            nxt: dict[tuple[int, int], float] = {}
            for (wins, losses), mass in alive.items():
                steps = ((1, 0, agent_only), (0, 1, searcher_only), (0, 0, 1 - agent_only - searcher_only))
                for more_wins, more_losses, chance in steps:
                    state = (wins + more_wins, losses + more_losses)
                    ratio = state[0] * win + state[1] * loss
                    reached = mass * chance
                    if ratio >= upper:
                        up += reached
                        expected += n * reached
                    elif ratio <= lower:
                        down += reached
                        expected += n * reached
                    elif n == n_max:
                        expected += n * reached
                    else:
                        nxt[state] = nxt.get(state, 0.0) + reached
            alive = nxt
        theta = agent_only / (agent_only + searcher_only)
        print(f"    {label:44s} theta {theta:.3f}: claim {up:.3f}, futility stop {down:.3f}, undecided {1 - up - down:.3f}, markets {expected:.0f}")


def study1() -> None:
    """The joint law of the two arms on one market is (agent's rate, G100's rate, both): the test's size holds
    for every joint law with the agent's rate at most G100's, and its power depends on all three."""
    joint = (
        ("null, every market discordant", 0.5, 0.5, 0.0),
        ("null 0.2, 0.2, independent", 0.2, 0.2, 0.04),
        ("0.5, 0.2, least overlap", 0.5, 0.2, 0.0),
        ("0.5, 0.2, independent", 0.5, 0.2, 0.1),
        ("0.5, 0.2, most overlap", 0.5, 0.2, 0.2),
        ("0.5, 0.25, least overlap", 0.5, 0.25, 0.0),
        ("0.5, 0.25, independent", 0.5, 0.25, 0.125),
        ("0.5, 0.15, least overlap", 0.5, 0.15, 0.0),
        ("0.5, 0.3, least overlap", 0.5, 0.3, 0.0),
        ("0.5, 0.3, independent", 0.5, 0.3, 0.15),
        ("0.5, 0.3, both 0.05 (Argon)", 0.5, 0.3, 0.05),
        ("0.5, 0.3, most overlap", 0.5, 0.3, 0.3),
        ("0.5, 0.1, independent", 0.5, 0.1, 0.05),
        ("0.35, 0.2, independent", 0.35, 0.2, 0.07),
        ("0.3, 0.2, independent", 0.3, 0.2, 0.06),
        ("0.25, 0.2, both 0.1875 (Argon)", 0.25, 0.2, 0.1875),
    )
    cases = tuple((label, agent - both, searcher - both) for label, agent, searcher, both in joint)
    for theta1, n_max in ((0.8, 60), (0.7, 100), (0.7, 400)):
        sign_test(n_max, theta1, cases)
    print("  the size is largest when every market is discordant; 400 markets stands for no truncation")
    print(f"  G100's rate near 0.2 on 400 calibration markets has standard error {math.sqrt(0.2 * 0.8 / 400):.3f}")
    print(f"  a pick whose effect is 4 standard errors from the threshold is misclassified with probability {norm.sf(4):.0e}")


def account(bets: int, cost: float, edge: float, fraction: float, weeks: int = 104, paths: int = 10_000) -> tuple[float, float]:
    """A self-financing paper account under the sequential rule: `bets` independent purchases a week at all-in
    cost `cost` and true probability cost*(1 + edge), each staking `fraction` of the cash left at that moment,
    all paid before the next week's purchases. Returns P(cash reaches 20) by week 52 and by `weeks`."""
    rng = np.random.default_rng(11)
    cash = np.ones(paths)
    by52 = np.zeros(paths, dtype=bool)
    reached = np.zeros(paths, dtype=bool)
    for week in range(weeks):
        payout = np.zeros(paths)
        for _ in range(bets):
            stake = fraction * cash
            cash = cash - stake
            payout += stake / cost * (rng.random(paths) < cost * (1 + edge))
        cash = cash + payout
        reached |= cash >= 20
        if week == 51:
            by52 = reached.copy()
    return float(by52.mean()), float(reached.mean())


def kelly(cost: float, edge: float) -> float:
    return cost * edge / (1 - cost)


def forward_power() -> None:
    """Power of a forward account in bounded bets, for planning: the stake is Kelly's fraction for the true edge,
    capped at 5% of cash (the best a calibrated agent can do), or half of it."""
    print("  columns: P(cash reaches 20 by week 52), by week 104")
    print(f"    no edge, 60 even bets a week at the cap: {account(60, 0.5, 0.0, STAKE_CAP)[1]:.3f} (the size; at most 0.05)")
    for bets in (5, 10, 15, 30, 60):
        for edge in (0.02, 0.04, 0.08, 0.12, 0.16):
            full = account(bets, 0.5, edge, min(STAKE_CAP, kelly(0.5, edge)))
            half = account(bets, 0.5, edge, min(STAKE_CAP, kelly(0.5, edge) / 2))
            print(f"    {bets:2d} even bets a week, net edge {edge:.2f}: Kelly capped {full[0]:.2f}, {full[1]:.2f}; half of it {half[0]:.2f}, {half[1]:.2f}")
    for bets in (5, 10, 15, 30, 60):
        found = brentq(lambda e: account(bets, 0.5, e, min(STAKE_CAP, kelly(0.5, e)))[1] - 0.8, 0.02, 0.6, xtol=0.002)
        print(f"    {bets:2d} even bets a week: 80% power within 104 weeks at a net edge of {found:.3f}")
    for edge in (0.115, 0.183):
        print(f"    on a return's sign, a net edge of {edge} is a hit rate of {(1 + edge) / 2:.3f}")
    for cost, points in ((0.2, 0.02), (0.2, 0.04), (0.8, 0.04)):
        edge = points / cost
        got = account(10, cost, edge, min(STAKE_CAP, kelly(cost, edge)))
        print(f"    10 bets a week at cost {cost}, {points} of probability (net edge {edge:.2f}): {got[0]:.2f}, {got[1]:.2f}")


def event_costs() -> None:
    """Kalshi's taker fee is round up(0.07 * C * P * (1 - P)) (fee schedule of 2026-07-07, audit/decisions.md).
    The half spread of one cent is an assumption, not a measurement."""
    half_spread = 0.01
    for mid in (0.2, 0.5, 0.8):
        ask = mid + half_spread
        fee = 0.07 * ask * (1 - ask)
        cost = ask + fee
        needs = ", ".join(f"net {edge:g}: {100 * (cost * (1 + edge) - mid):.1f}" for edge in (0.0, 0.04, 0.09, 0.115, 0.183))
        print(f"  mid {mid}: fee {100 * fee / ask:.1f}% of the stake, all-in cost {cost:.4f}; points of probability above the mid for {needs}")
    print(f"  $125 at 0.50 is {125 / 0.5:.0f} contracts, at 0.20 {125 / 0.2:.0f}; fee on 250 at 0.50: ${math.ceil(7 * 250 * 0.25) / 100:.2f}")


def log_growth(chance: float, gain: float, loss: float) -> float:
    return chance * math.log(1 + gain) + (1 - chance) * math.log(1 - loss)


def late_records(hold: int, bets: int = 400, paths: int = 200_000) -> float:
    """A fair account (every purchase at cost 0.05 with probability 0.05, staking 5% of recorded cash) whose every
    payout is recorded only `hold` purchases late, as in Argon's finding 5. Returns P(recorded cash reaches 20)."""
    rng = np.random.default_rng(5)
    cash = np.ones(paths)
    pending = np.zeros((paths, hold + 1))
    reached = np.zeros(paths, dtype=bool)
    for bet in range(bets):
        slot = bet % (hold + 1)
        cash = cash + pending[:, slot]
        reached |= cash >= 20
        stake = STAKE_CAP * cash
        cash = cash - stake
        pending[:, slot] = stake / 0.05 * (rng.random(paths) < 0.05)
    return float(reached.mean())

def expected_max(count: int) -> float:
    """The expected maximum of `count` independent standard Gaussians, by quadrature."""
    value, _ = quad(lambda x: x * count * norm.pdf(x) * norm.cdf(x) ** (count - 1), -10, 12, limit=200)
    return float(value)


def dsr_z(sharpe: float, threshold: float, steps: int, skew: float, kurtosis: float) -> float:
    spread = math.sqrt(1 - skew * sharpe + (kurtosis - 1) / 4 * sharpe**2)
    return (sharpe - threshold) * math.sqrt(steps - 1) / spread


def argon_paper() -> None:
    """Rechecks of Argon's findings on the paper (audit/program/argon-paper-findings.md)."""
    print("  4. Clopper-Pearson upper bounds at 95%, no promotion in n:")
    for n in (9, 10, 13, 21):
        print(f"       n = {n}: {1 - 0.05 ** (1 / n):.3f}")
    rho = 8 / math.sqrt(46.25 * 2.6)
    print(f"  5. correlation {rho:.4f}; a null survivor at z = 3 passes the sign check with probability {norm.cdf(3 * rho / math.sqrt(1 - rho**2)):.4f}")
    print(f"  6. Hoeffding cutoff {0.2 - math.sqrt(math.log(20) / 200):.4f}")
    year = math.sqrt(STEPS_A_YEAR)
    for steps in (100, 546, 5400):
        print(f"  7. T = {steps}: the limit of the DSR's z as the Sharpe grows is under 1.645 only above kurtosis {1 + 4 * (steps - 1) / norm.ppf(0.95) ** 2:.0f}; a sample's is at most {steps}")
    print(f"     Argon's examples at T = 546 against a threshold of 2: Sharpe 3.5 has z at most {dsr_z(3.5 / year, 2 / year, 546, 0, 1):.3f}; Sharpe 6, skewness -4, kurtosis 18 has z = {dsr_z(6 / year, 2 / year, 546, -4, 18):.3f}; the limit at kurtosis 18 is {2 * math.sqrt(545 / 17):.1f}")
    for level, tolerance, inflated in ((0.05, 0.0625, 0.056), (0.01, 0.015, 0.012), (0.005, 0.01, 0.0075)):
        suite = max(x for x in range(2000) if beta.ppf(1 - 0.05 / 99, x + 1, 20_000 - x) <= tolerance)
        single = max(x for x in range(2000) if beta.ppf(0.95, x + 1, 20_000 - x) <= tolerance)
        chance = binom.cdf(suite, 20_000, inflated)
        print(f"  10. level {level}, tolerance {tolerance}: up to {suite} rejections in 20,000 pass at the suite's confidence 1 - 0.05/99 ({single} at a single cell's 95%); a true rate of {inflated} passes with probability {chance:.2f}")
    phi, keep = 1 - 1 / 42, 5 / 6
    whole, kept = (1 + phi) / (1 - phi), (1 + phi * keep) / (1 - phi * keep)
    print(f"  11. a funding leg's long-run variance factor is {whole:.1f}, of which blocks of mean 6 keep {kept:.2f}")
    for share in (0.000025, 0.001, 0.01, 0.1):
        print(f"      funding at {share:g} of a step's variance: the bootstrap understates the variance of the mean by {(1 - share + share * whole) / (1 - share + share * kept):.3f}")
    share = brentq(lambda w: (1 - w + w * whole) / (1 - w + w * kept) - 1.1, 0, 1)
    print(f"      the understatement reaches 10% at a share of {share:.4f}: a funding standard deviation of {100 * math.sqrt(share) * 0.02:.3f}% a step beside 2% of price")
    print("  13. expected maximum of N Gaussians against sqrt(2 log N), and the ceiling at 3, 10, 20 years:")
    for count in (10, 1000, 100_000):
        exact = expected_max(count)
        ceilings = ", ".join(f"{exact / math.sqrt(years):.2f}" for years in (3, 10, 20))
        print(f"       N = {count}: {exact:.4f} against {math.sqrt(2 * math.log(count)):.4f}; {ceilings}")
    se = math.sqrt(365 / 550)
    print(f"  14. standard error {se:.2f}; a Sharpe 2 leak stays under the canary's alarm at 4 with probability {norm.cdf((4 - 2) / se):.3f}")
