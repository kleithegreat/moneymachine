def yardstick() -> None:
    """The Gaussian fixed-horizon yardstick, z = S * sqrt(T): a planning reference, not a limit on every test."""

def crossing(drift: float, spread: float, level: float = LOG20) -> float:
    """P(a Brownian log wealth with total drift `drift` and total standard deviation `spread` reaches `level`
    at any time before the end): the reflection formula. The test stops there, so this is its power."""
    reflected = math.exp(min(0.0, 2 * drift * level / spread**2 + norm.logcdf((-drift - level) / spread)))
    return float(norm.cdf((drift - level) / spread)) + reflected


def kelly_power(u: float) -> float:
    """Power of an exact test staked at Kelly's fraction, where u = effect * sqrt(count)."""
    return crossing(u * u / 2, u)


U80 = brentq(lambda u: kelly_power(u) - 0.8, 1, 6)

def account(bets: int, hold: int, edge: float, fraction: float, weeks: int = 104, paths: int = 4000) -> float:
    """A paper account in even bets with net edge `edge`: `bets` purchases a week, each staking `fraction` of the
    cash left at that moment and paid `hold` purchases later (0: before the next purchase). Returns P(cash reaches
    20 within `weeks`); a payout still pending at the end is not cash (Argon's round 4, finding 7)."""
    rng = np.random.default_rng(11)
    cash = np.ones(paths)
    pending = np.zeros((paths, hold + 1))
    reached = np.zeros(paths, dtype=bool)
    for bet in range(weeks * bets):
        slot = bet % (hold + 1)
        cash = cash + pending[:, slot]
        reached |= cash >= 20
        stake = fraction * cash
        cash = cash - stake
        pending[:, slot] = 2 * stake * (rng.random(paths) < (1 + edge) / 2)
    return float(reached.mean())

def capped_growth(mu: float, sigma: float, cap: float) -> float:
    """Expected log growth a step of wealth bet at the constant fraction `cap`, to second order."""
    return cap * mu - cap**2 * (sigma**2 + mu**2) / 2


def series_power(sharpe: float, planned: tuple[float, ...], paths: int = 20_000) -> tuple[float, float]:
    """A frozen rule's four-hour returns with annualized Sharpe `sharpe`, Gaussian. The confirmer's wealth is the
    mean of the wealths that bet Kelly's fraction for each planned Sharpe. Returns P(wealth reaches 20) within one
    year and within two."""
    rng = np.random.default_rng(7)
    dt = 1 / STEPS_A_YEAR
    wealth = np.ones((len(planned), paths))
    bets = np.array(planned)[:, None]
    reached = np.zeros(paths, dtype=bool)
    first = reached
    for step in range(2 * STEPS_A_YEAR):
        wealth = wealth * (1 + bets * rng.normal(sharpe * dt, math.sqrt(dt), paths))
        reached |= wealth.mean(axis=0) >= 20
        if step == STEPS_A_YEAR - 1:
            first = reached.copy()
    return float(first.mean()), float(reached.mean())


def confirmer(sigma: float = 0.02) -> None:
    """A frozen return series under three confirmers: the yardstick, the gate's test (bet at most 1/L, L = 2), and
    an exact test whose loss bound is the rule's own, so that Kelly's fraction fits (Argon's finding 2)."""
    annual = math.sqrt(STEPS_A_YEAR)
    print("  Sharpe for 80% power: yardstick; exact test at Kelly's fraction; the gate's capped test")
    for years in (91 / 365, 1, 2):
        steps = round(years * STEPS_A_YEAR)
        spread = 0.5 * sigma * math.sqrt(steps)
        capped = brentq(lambda m: crossing(steps * capped_growth(m, sigma, 0.5), spread) - 0.8, 0, 1)
        print(f"    {steps} steps: {Z80 / math.sqrt(years):.2f}; {U80 / math.sqrt(years):.2f}; {capped * annual / sigma:.1f}")
    for sharpe in (1.4, 2, 2.4):
        print(f"  Kelly's fraction fits under 1/L while the worst step loss L is at most {annual / sharpe:.1f} step standard deviations at Sharpe {sharpe}; the gate's L is {2 / sigma:.0f}")
    print("  power within one year and two, by true Sharpe; betting Kelly's fraction for Sharpe 2 (closed form at two years), and the mean of the bets for 1, 2 and 3:")
    for sharpe in (0, 1, 1.4, 2, 2.4, 3):
        fixed = series_power(sharpe, (2.0,))
        mixed = series_power(sharpe, (1.0, 2.0, 3.0))
        closed = crossing((2 * sharpe - 2) * 2, 2 * math.sqrt(2))
        print(f"    Sharpe {sharpe}: for 2, {fixed[0]:.2f} and {fixed[1]:.2f} ({closed:.2f}); mean of three, {mixed[0]:.2f} and {mixed[1]:.2f}; at Kelly's own fraction {kelly_power(sharpe * math.sqrt(2)) if sharpe else 0:.2f} at two years")
    print(f"  the standard error of a Sharpe measured over two years is {1 / math.sqrt(2):.2f}")
    for sharpe in (1.4, 2.4):
        print(f"  years to 80% power at Sharpe {sharpe}: {U80**2 / sharpe**2:.2f} at Kelly's fraction")

def years_to_power(sharpe: float, vol: float, power: float = 0.8) -> float:
    """Years until an account of yearly volatility `vol` and Sharpe `sharpe`, taken as the wealth, has reached 20
    with probability `power`. Infinite when it never does."""
    drift = sharpe * vol - vol**2 / 2

    def shortfall(years: float) -> float:
        return crossing(drift * years, vol * math.sqrt(years)) - power

    return brentq(shortfall, 0.01, 1000) if shortfall(1000) > 0 else math.inf


def dual_bound(first: np.ndarray, second: np.ndarray, third: np.ndarray) -> float:
    """The confirmer's bound on the log wealth of every constant mix, from three sums a coin, at the best
    multiplier. Since log(1 + x) <= x - x^2/2 + x^3/3 for x >= -1, a mix's log wealth is at most
    sum_c b_c S1_c - b_c^2 Q_c with Q_c = S2_c/2 - max(S3_c, 0)/3, and for any multiplier lam >= 0 the maximum
    over mixes is at most D(lam) = lam + sum_c max over b in [0, 1] of (S1_c - lam) b - Q_c b^2, which is closed
    form. Every coin's term is zero once lam >= S1_c - min(Q_c, 0), and D rises with lam from there, so the
    least D lies below the largest of those (Argon's finding 4 of round 5: the search stopped at the largest S1)."""
    curve = second / 2 - np.clip(third, 0, None) / 3

    def dual(multiplier: float) -> float:
        slope = first - multiplier
        inner = np.where(curve > 0, np.clip(slope / (2 * np.where(curve > 0, curve, 1)), 0, 1), (slope > curve).astype(float))
        return multiplier + float((slope * inner - curve * inner**2).sum())

    top = max(0.0, float((first - np.minimum(curve, 0)).max()))
    return float(minimize_scalar(dual, bounds=(0, top + 1), method="bounded", options={"xatol": 1e-12}).fun)


def mix_bound(returns: np.ndarray) -> tuple[float, float]:
    """The best constant mix in hindsight on the record's own clock, and the bound the confirmer certifies.
    `returns` holds each coin's quote-to-quote simple returns in a column; a mix b (long only, summing to at
    most 1) rebalances at every quote, so its log wealth is the sum over coins of sum log(1 + b_c r). Returns
    the log wealth of the mix an optimizer found and `dual_bound` of the three sums."""
    bound = dual_bound(returns.sum(axis=0), (returns**2).sum(axis=0), (returns**3).sum(axis=0))
    coins = returns.shape[1]
    found = minimize(
        lambda b: -np.log1p(returns * b).sum(),
        np.full(coins, 1 / (coins + 1)),
        jac=lambda b: -(returns / (1 + returns * b)).sum(axis=0),
        bounds=[(0, 1)] * coins,
        constraints={"type": "ineq", "fun": lambda b: 1 - b.sum(), "jac": lambda b: -np.ones(coins)},
        method="SLSQP",
        options={"maxiter": 500, "ftol": 1e-13},
    )
    mix = np.clip(found.x, 0, None)
    mix = mix / max(1.0, mix.sum())
    return float(np.log1p(returns * mix).sum()), float(bound)


def hindsight_cost(drift: float, paths: int = 200, days: int = 730, coins: int = 16, vol: float = 0.8, shared: float = 0.6) -> tuple[float, float, float]:
    """The log wealth of the best constant mix in hindsight over two years: `coins` coins of yearly volatility
    `vol`, a share `shared` of each one's variance common, every coin's yearly drift `drift`, one quote a coin a
    day (assumptions, not measurements; a diffusion's best mix barely depends on the clock, see
    `rebalancing_residual`). Returns the median and the 0.9 quantile of the certified bound, and the widest gap
    between the bound and the mix found."""
    rng = np.random.default_rng(13)
    step = vol / math.sqrt(365)
    bounds, gaps = [], []
    for _ in range(paths):
        common = rng.normal(0, step * math.sqrt(shared), (days, 1))
        own = rng.normal(0, step * math.sqrt(1 - shared), (days, coins))
        returns = np.expm1((drift - vol**2 / 2) / 365 + common + own)
        log_wealth, bound = mix_bound(returns)
        bounds.append(bound)
        gaps.append(bound - log_wealth)
    return float(np.median(bounds)), float(np.quantile(bounds, 0.9)), max(gaps)


def rebalancing_residual(drift: float = 0.3, vol: float = 0.8) -> float:
    """Yearly log growth of the best constant exposure rebalanced continuously, less that of the best rebalanced
    daily, in one lognormal coin: what a daily clock hides in a diffusion. It is not a bound: `round3` shows a
    market of jumps where it is 0.006 a day."""
    step = vol / math.sqrt(365)
    mean = (drift - vol**2 / 2) / 365

    def daily(exposure: float) -> float:
        return quad(lambda x: math.log1p(exposure * math.expm1(x)) * norm.pdf(x, mean, step), mean - 10 * step, mean + 10 * step)[0]

    best_daily = -minimize_scalar(lambda e: -daily(e), bounds=(0, 1), method="bounded").fun * 365
    exposure = min(1.0, max(0.0, drift / vol**2))
    return exposure * drift - exposure**2 * vol**2 / 2 - best_daily

def forecast_ratio(returns: np.ndarray, record: int, forward: int, block: int, draws: int, rng: np.random.Generator) -> np.ndarray:
    """Draws of (the variance of a pseudo-forward of `forward` days) over (that of a pseudo-record of `record` days), each made of circular blocks of
    `block` days of `returns` and each about its own mean, divided by its count: study 2's bootstrap for CS2.2."""
    days = len(returns)

    def variances(length: int) -> np.ndarray:
        starts = rng.integers(0, days, size=(draws, math.ceil(length / block)))
        index = ((starts[:, :, None] + np.arange(block)[None, None, :]) % days).reshape(draws, -1)[:, :length]
        return returns[index].var(axis=1)

    return variances(forward) / variances(record)


def clustered_days(days: int, persistence: float, rng: np.random.Generator) -> np.ndarray:
    """Daily returns whose variance follows a GARCH(1,1) of the given persistence, a tenth of it from the last
    return's square, started from its stationary mean after a burn-in of 500 days."""
    reaction = 0.1
    variance, out = 1.0, np.empty(days + 500)
    for day in range(days + 500):
        out[day] = math.sqrt(variance) * rng.standard_normal()
        variance = (1 - persistence) + reaction * out[day] ** 2 + (persistence - reaction) * variance
    return out[500:]


def held(replay: list[Fraction], forward: list[Fraction]) -> str:
    """CS2.2's verdict from two lists of daily returns, exact: each list's variance is about its own mean and divided by its count, and the two are
    compared by cross-multiplying. A replay of no variance has no ratio."""

    def scaled(returns: list[Fraction]) -> Fraction:
        return len(returns) * sum(r * r for r in returns) - sum(returns) ** 2

    below, above = scaled(replay) * len(forward) ** 2, scaled(forward) * len(replay) ** 2
    if below == 0:
        return "not judged"
    return "held" if below <= 2 * above and above <= 2 * below else "did not hold"


def dual_share(s1: Fraction, q: Fraction, multiplier: Fraction) -> Fraction:
    """The weight in [0, 1] at which one coin's term of the bar's bound, (s1 - multiplier) b - q b^2, is greatest; a tie goes to the lesser."""
    if q > 0:
        return min(Fraction(1), max(Fraction(0), (s1 - multiplier) / (2 * q)))
    return Fraction(int(s1 - multiplier > q))


def least_dual(first: list[Fraction], curve: list[Fraction]) -> tuple[Fraction, Fraction]:
    """The bar's bound D at the multiplier Lean computes for it, and that multiplier (architecture 6.1, the certificate). D is convex in the
    multiplier, and between two neighbouring points where a coin's term changes form it is a quadratic. So its least over every multiplier is its
    least over zero, those points above zero, and between two neighbours the point where its slope is zero when there is one: a finite list of
    rationals, each exact. Ties go to the least multiplier."""

    def dual(multiplier: Fraction) -> Fraction:
        shares = [dual_share(s1, q, multiplier) for s1, q in zip(first, curve)]
        terms = ((s1 - multiplier) * share - q * share**2 for s1, q, share in zip(first, curve, shares))
        return multiplier + sum(terms, Fraction(0))

    coins = list(zip(first, curve))
    changes = {point for s1, q in coins for point in ((s1, s1 - 2 * q) if q > 0 else (s1 - q,)) if point > 0}
    points = sorted(changes | {Fraction(0)})
    candidates = list(points)
    for low, high in zip(points, points[1:]):
        middle = (low + high) / 2
        partial = [(s1, q) for s1, q in coins if q > 0 and 0 < s1 - middle < 2 * q]
        whole = sum(1 for s1, q in coins if dual_share(s1, q, middle) == 1)
        if partial:
            flat = (sum((s1 / (2 * q) for s1, q in partial), Fraction(0)) + whole - 1) / sum((1 / (2 * q) for _, q in partial), Fraction(0))
            if low < flat < high:
                candidates.append(flat)
    return min((dual(multiplier), multiplier) for multiplier in candidates)


def bar_mix(first: list[Fraction], curve: list[Fraction]) -> list[Fraction]:
    """The bar's own mix: each coin's weight where its term of the bound is greatest at the least multiplier, all divided by their sum when that is
    above 1. It is the guard's second mix, and nothing searches for it."""
    _, multiplier = least_dual(first, curve)
    shares = [dual_share(s1, q, multiplier) for s1, q in zip(first, curve)]
    return [share / max(Fraction(1), sum(shares, Fraction(0))) for share in shares]


def day_slices(rows: list[tuple[str, Fraction]]) -> list[Fraction]:
    """What each day added to one of a coin's running fields (a sum, the funding a unit paid, the count of events), from the coin's rows in the
    report's order, each (occasion, the field since its span began): KERNEL.md A19's `coin` rows. A row adds its value less that of the coin's row
    before it in the same span, or its whole value where it is its span's first, and a day's slice is what the rows after the day before's row, up
    to its own, added. Brief 4's step 5 chains these, never the rows themselves (GPT's R2)."""
    slices, carried, adding = [], Fraction(0), Fraction(0)
    for occasion, value in rows:
        adding += value - carried
        carried = value
        if occasion == "span":
            carried = Fraction(0)
        else:
            slices.append(adding)
            adding = Fraction(0)
    assert adding == 0, "a span's row after the last day's repeats that day's"
    return slices


def bootstrap_days(record: int, paths: int, block: int = 14, days: int = 730) -> np.ndarray:
    """Brief 4's step 5: for each path, the record's day (from 0) at each of its `days` places. A path is blocks of `block` consecutive days, the
    last cut, each running past the record's last day to its first; block j of path i starts at the BLAKE2b-256 digest of `study2 bootstrap i j`,
    as one whole number, modulo the record's days."""
    blocks = math.ceil(days / block)

def reach(market: list[tuple[int, str, Fraction]], funding: list[tuple[int, str, Fraction]], weights: dict[str, Fraction], clock: int, minutes: int) -> tuple[list[float], bool]:
    """A guard rule's wealth over the bar at each day's end of a record, and whether it ended. `funding` holds (time, coin, what a unit paid); the
    bar here is of the prices alone, which a small charge moves by its own size over the price."""
    events = sorted([(time, "price", coin, new) for time, coin, new in market] + [(time, "funding", coin, paid) for time, coin, paid in funding])
    rule = GuardRule(weights)
    wealth_at = guard_fold(rule, events, clock, minutes * MINUTE, {"A": Fraction(100), "B": Fraction(100)})
    return [float(wealth) / math.exp(bar_log(market, day_end)) for day_end, wealth in sorted(wealth_at.items())], rule.ended


def refused(rules: list[list[Fraction]]) -> bool:
    """CS2.1's refusal of rung 2, exact, from each guard rule's wealth over the bar at each day's end of the replay: some rule stood 20 times over
    the bar at a day's end, or stands at the last so far over it that the same rate reaches 20 within the 730 judged days."""
    return any(max(ratios) >= 20 or ratios[-1] ** 730 >= 20 ** len(ratios) for ratios in rules)

        price = first = 100.0 * 1.0001
        cash = 1.0
        scale = share * cash / sum(LADDER_NOTIONAL[:depth])
        spend = min(scale * LADDER_NOTIONAL[0] * (1 + TAKER_FEE), cash)
        cash -= spend
        test_units = spend / (first * (1 + TAKER_FEE))
        units, cost, level = LADDER_NOTIONAL[0] / first, LADDER_NOTIONAL[0], 1
        price = 100.0
        for instant in range(limit):
            price *= math.exp(sigma * rng.standard_normal())
            while level < len(LADDER_NOTIONAL) and price < first * (1 - LEVEL_GAP * level):
                level_price = first * (1 - LEVEL_GAP * level)
                spend = min(scale * LADDER_NOTIONAL[level] * (1 + MAKER_FEE), cash) if level < depth else 0.0
                cash -= spend
                test_units += spend / (level_price * (1 + MAKER_FEE))
                units, cost, level = units + LADDER_NOTIONAL[level] / level_price, cost + LADDER_NOTIONAL[level], level + 1
            exit_price = cost / units * (1 + TAKE_PROFIT)
            if price > exit_price:
                cash += test_units * exit_price * (1 - MAKER_FEE)
                break
        else:
            cash += test_units * price * (1 - TAKER_FEE)
        gains.append(cash - 1)
    return statistics.mean(gains), statistics.stdev(gains) / math.sqrt(cycles), statistics.stdev(gains)

def least_power(value: float) -> int:
    """The least of 1, 2, 4, ... at or above `value`: 1 for every value up to 1, zero included."""
    power = 1
    while power < value:
        power *= 2
    return power


TEST_LOT = Fraction(1, 10**30)  # the test ledger's own lot, of any coin
CASH_GRID = TEST_LOT / (2 * 10**20)  # a price's scale of 8, a rate's of 12 (kernel A2), and a mid's half


def whole_lots(units: Fraction) -> Fraction:
    return (units // TEST_LOT) * TEST_LOT


class Books:
    """Cash, from 1, and units of coins in whole test lots, neither ever negative, valued at prices. Funding is paid from the cash, and a payment the
    cash cannot meet ends the books. The test ledger and the guard's rules are both these books, and differ only in what makes them trade."""

    def __init__(self) -> None:
        self.cash = Fraction(1)
        self.units: dict[str, Fraction] = {}
        self.ended = False

    def fund(self, coin: str, unit_payment: Fraction) -> None:
        payment = self.units.get(coin, Fraction(0)) * unit_payment
        if payment > self.cash:
            self.ended = True
        else:
            self.cash -= payment

    def wealth(self, prices: dict[str, Fraction]) -> Fraction:
        return self.cash + sum((held * prices[coin] for coin, held in self.units.items()), Fraction(0))

    def wealth_at_bid(self, bids: dict[str, Fraction], prices: dict[str, Fraction]) -> Fraction:
        """What the rungs compare (architecture 6.1): each coin at its bid price, the lesser of its latest quote's best bid and its price."""
        return self.cash + sum((held * min(bids[coin], prices[coin]) for coin, held in self.units.items()), Fraction(0))


class TestLedger(Books):
    """The test ledger of architecture 6.1 and kernel A12, exact. It trades only at the account's fills, at their prices and fee rates. A cycle's scale
    is fixed at its first buy as `share` of the ledger's cash then over `depth`, a sum in the account's money; the ledger copies the cycle's first
    `depth` of the account's buys at that scale, each cut to its cash and down to whole lots, and nothing after. No number in it is rounded but a
    trade's size, so there is no second, approximate ledger to compare it with."""

    def __init__(self, share: Fraction, depth: Fraction) -> None:
        super().__init__()
        self.share, self.depth = share, depth
        self.scale: dict[str, Fraction] = {}
        self.followed: dict[str, Fraction] = {}

    def buy(self, coin: str, lots: Fraction, price: Fraction, fee: Fraction) -> None:
        if coin not in self.scale:
            self.scale[coin] = self.share * self.cash / self.depth
            self.followed[coin] = Fraction(0)
        notional = lots * price
        within = max(Fraction(0), min(notional, self.depth - self.followed[coin]))
        self.followed[coin] += notional
        cost = price * (1 + fee)
        bought = whole_lots(min(self.scale[coin] * within / price, self.cash / cost))
        self.cash -= bought * cost
        self.units[coin] = self.units.get(coin, Fraction(0)) + bought

    def sell(self, coin: str, fraction: Fraction, price: Fraction, fee: Fraction) -> None:
        held = self.units.get(coin, Fraction(0))
        sold = held if fraction == 1 else whole_lots(held * fraction)
        self.cash += sold * price * (1 - fee)
        self.units[coin] = held - sold
        if fraction == 1:
            del self.scale[coin], self.followed[coin]


class GuardRule(Books):
    """One rule of rung 2's guard (architecture 6.1): a constant mix held in the books above and traded back to its weights at its clock's times, at
    prices and at no cost. A trade at the price moves no wealth, so the rule is of the test ledger's kind and uses nothing."""

    def __init__(self, weights: dict[str, Fraction]) -> None:
        super().__init__()
        assert all(weight >= 0 for weight in weights.values()) and sum(weights.values()) <= 1
        self.weights = weights

    def rebalance(self, prices: dict[str, Fraction]) -> None:
        wealth = self.wealth(prices)
        self.units = {coin: whole_lots(weight * wealth / prices[coin]) for coin, weight in self.weights.items()}
        self.cash = wealth - sum((held * prices[coin] for coin, held in self.units.items()), Fraction(0))

def pairs_market(period: int, minutes: int, astride: bool = False) -> list[tuple[int, str, Fraction]]:
    """Two coins quoted in turn, as price events (time in ms, coin, new price) in the machine's order. In a pair A moves and then B, one up a quarter
    and the other down a fifth, the two changing places from one pair to the next, so A's price is 100 or 125 and B's 100 or 80 for ever: on the
    venue's grid of five significant figures and inside its range, which Argon's 0.25 and -0.15 left within three moves (GPT's finding 4, round 6).
    A pair comes every `period` minutes, A in its first minute and B in its second; or, `astride`, every minute, A a millisecond before a tick and B
    a millisecond after it."""
    events = []
    spacing = MINUTE if astride else period * MINUTE
    for pair in range(minutes * MINUTE // spacing):
        a_rises = pair % 2 == 0
        a_time = (pair + 1) * MINUTE - 1 if astride else pair * spacing + MINUTE // 2
        b_time = (pair + 1) * MINUTE + 1 if astride else pair * spacing + 3 * MINUTE // 2
        events.append((a_time, "A", Fraction(125 if a_rises else 100)))
        events.append((b_time, "B", Fraction(80 if a_rises else 100)))
    return sorted(event for event in events if event[0] < minutes * MINUTE)


def bar_log(events: list[tuple[int, str, Fraction]], until: int) -> float:
    """The log of the bar at the day's end `until`, from what that day's coin rows hold (kernel A12, K2; KERNEL.md A19): each coin's three sums over
    its price events before `until`, and its price over its first. One span, no funding."""
    price = {"A": Fraction(100), "B": Fraction(100)}
    sums = {coin: np.zeros(3) for coin in price}
    for time, coin, new in events:
        if time >= until:
            break
        change = float(new / price[coin] - 1)
        sums[coin] += [change, change**2, change**3]
        price[coin] = new
    first, second, third = (np.array([sums[coin][power] for coin in "AB"]) for power in range(3))
    alone = max(float(now) / 100 for now in price.values())
    return max(dual_bound(first, second, third), math.log(alone))


def guard_fold(rule: GuardRule, events: list[tuple[int, str, str, Fraction]], clock: int, until: int, first: dict[str, Fraction]) -> dict[int, Fraction]:
    """Kernel B1's `guard`: the rule's books folded over the engine's price and funding events and a minute's ticks in the machine's order, a tick
    after every event of its timestamp. `events` are (time, kind, coin, value): a "price" sets the coin's price, and a "funding" is paid, as the
    ledger pays it, on the units held into it, where the first the cash cannot meet ends the rule. The rule sees prices only at ticks, and trades
    back to its mix at those that are whole multiples of `clock` minutes. Returns its wealth at each day's end, zero once it has ended. A day's end
    is KERNEL.md A19's, the world after every event before the midnight and before any stamped with it, so that a run cut back to a midnight gives
    that day what the longer run gave it (GPT's R1 and Argon's F2 on study 2's criteria)."""
    price = dict(first)
    wealth_at: dict[int, Fraction] = {}
    folded = 0

    def fold(through: int) -> None:
        nonlocal folded
        while folded < len(events) and events[folded][0] <= through:
            _, kind, coin, value = events[folded]
            if kind == "price":
                price[coin] = value
            elif not rule.ended:
                rule.fund(coin, value)
            folded += 1

    for tick in range(0, until + 1, MINUTE):
        if tick > 0 and tick % DAY == 0:
            fold(tick - 1)
            wealth_at[tick] = Fraction(0) if rule.ended else rule.wealth(price)
        fold(tick)
        if not rule.ended and tick % (clock * MINUTE) == 0:
            rule.rebalance(price)
            assert rule.cash >= 0 and (rule.cash / CASH_GRID).denominator == 1
    return wealth_at


def guard_reach(market: list[tuple[int, str, Fraction]], clock: int, minutes: int) -> dict[int, float]:
    """The most that a mix on a grid of tenths, rebalanced every `clock` minutes, stood over the bar at each day's end: round 6's fixtures. The
    criteria's guard searches nothing (bar_mix, study2_criteria)."""
    events = [(time, "price", coin, new) for time, coin, new in market]
    most: dict[int, float] = {}
    for tenths_a in range(11):
        for tenths_b in range(11 - tenths_a):
            rule = GuardRule({"A": Fraction(tenths_a, 10), "B": Fraction(tenths_b, 10)})
            start = {"A": Fraction(100), "B": Fraction(100)}
            for day_end, wealth in guard_fold(rule, events, clock, minutes * MINUTE, start).items():
                over = math.exp(math.log(wealth) - bar_log(market, day_end))
                most[day_end // DAY] = max(most.get(day_end // DAY, 0.0), over)
    return most


def grid_step(price: Fraction, figures: int, decimals: int) -> Fraction:
    """The price grid's step at a price above zero (KERNEL.md A18): for a price in [10^e, 10^(e+1)) it is 10^max(min(e - figures + 1, 0), -decimals)."""
    exponent = 0
    while Fraction(10) ** (exponent + 1) <= price:
        exponent += 1
    while Fraction(10) ** exponent > price:
        exponent -= 1
    return Fraction(10) ** max(min(exponent - figures + 1, 0), -decimals)


def to_grid(price: Fraction, figures: int, decimals: int, up: bool) -> Fraction:
    """A sell's price rounds up to the grid and a buy's down (kernel A2). At or under zero there is no grid price at or below: a buy's is zero and a
    sell's the least on the grid (KERNEL.md A18; Argon's note 2)."""
    if price <= 0:
        return Fraction(10) ** -decimals if up else Fraction(0)
    step = grid_step(price, figures, decimals)
    return (math.ceil(price / step) if up else math.floor(price / step)) * step


def on_grid(price: Fraction, figures: int, decimals: int) -> bool:
    """A18's grid as its sentence says it, with no step: above zero, a whole multiple of 10^-decimals, and of at most `figures` significant figures or
    a whole number. For a decimal of at most 8 places, a price's scale."""
    scaled = price * 10**8
    assert scaled.denominator == 1
    digits = str(scaled.numerator).rstrip("0")
    multiple = (price * Fraction(10) ** decimals).denominator == 1
    return price > 0 and multiple and (len(digits) <= figures or price.denominator == 1)


def price_grid() -> None:
    """The venue's certified grid (venue/sources.toml, "Price grid"): its two examples, the edge of five figures, and on every price of a range that
    the sentence and the step are one grid and the two roundings are its neighbours."""
    cases = {"123456": True, "12345.6": False, "99999": True, "100000": True, "100001": True, "100000.5": False, "9999.9": True, "0.5": True}
    for text, expected in cases.items():
        assert on_grid(Fraction(text), 5, 1) is expected, text
    for text, down, up in (("100000.5", 100_000, 100_001), ("12345.6", 12_345, 12_346), ("99999.5", 99_999, 100_000)):
        assert to_grid(Fraction(text), 5, 1, up=False) == down and to_grid(Fraction(text), 5, 1, up=True) == up, text
    without_whole_numbers = Fraction(10) ** max(5 - 5 + 1, -1)
    assert Fraction(100_001) % without_whole_numbers != 0
    checked = 0
    for figures, decimals_tried, top in ((1, (-1, 0, 1, 2), 12_000), (2, (-1, 0, 1, 2), 12_000), (3, (0, 1), 120_000)):
        for decimals in decimals_tried:
            prices = [Fraction(hundredths, 100) for hundredths in range(1, top + 1)]
            members = [price for price in prices if on_grid(price, figures, decimals)]
            place = 0
            for price in prices:
                while place < len(members) and members[place] < price:
                    place += 1
                step = grid_step(price, figures, decimals)
                assert ((price / step).denominator == 1) == (place < len(members) and members[place] == price), (price, figures, decimals)
                below = members[place] if place < len(members) and members[place] == price else members[place - 1] if place else Fraction(0)
                assert to_grid(price, figures, decimals, up=False) == below, (price, figures, decimals)
                if place < len(members):
                    assert to_grid(price, figures, decimals, up=True) == members[place], (price, figures, decimals)
                checked += 1
    print(f"  the certificate's two examples at five figures and one decimal: 123456 is a price and 12345.6 is not, a buy there rounding to 12345 and a sell to 12346. At the edge of five figures 99,999, 100,000 and 100,001 are prices and 100,000.5 is not; the step without the whole numbers was {without_whole_numbers} from 100,000 up and refused 100,001")
    print(f"  the sentence and the step are one grid: on {checked:,} prices in hundredths, at one to three figures and from -1 to 2 decimals, a price is a whole multiple of its step exactly when it is on the grid as the sentence says, a buy rounds to the greatest grid price at or below it (zero when there is none) and a sell to the least at or above")


def construction_a(bid_follows: bool) -> tuple[Fraction, Fraction, float, Fraction]:
    """Argon's construction A on study 2's criteria, at prices the engine admits (five figures, four decimals). One coin is quoted down from 100 to 4
    in a thousand quotes. The ledger, at phi = 1 and a depth of the buy's notional, buys at the ask of 4.0002 with the taker's fee. Then 18 quotes
    each raise the mid by two fifths with no print: the ask alone, the bid staying at 4, or with `bid_follows` both sides, the bid a thousandth under
    the mid. Returns the wealth at prices, the wealth at the bid, the bar and the last bid, at that day's end."""
    figures, decimals, taker = 5, 4, Fraction(45, 100_000)
    quotes: list[tuple[Fraction, Fraction]] = []
    for step in range(1000):
        centre = Fraction(100 * 0.04 ** (step / 1000))
        low = to_grid(centre, figures, decimals, up=False)
        quotes.append((low, low + grid_step(low, figures, decimals)))
    floor = Fraction(4)
    quotes.append((floor, Fraction(40_002, 10_000)))
    ledger = TestLedger(Fraction(1), Fraction(100))
    ledger.buy("A", Fraction(100) / quotes[-1][1], quotes[-1][1], taker)
    for pull in range(1, 19):
        mid = floor * Fraction(7, 5) ** pull
        if bid_follows:
            quotes.append((to_grid(mid * Fraction(999, 1000), figures, decimals, up=False), to_grid(mid * Fraction(1001, 1000), figures, decimals, up=True)))
        else:
            quotes.append((floor, to_grid(2 * mid - floor, figures, decimals, up=True)))
    assert all(0 < bid < ask and bid % grid_step(bid, figures, decimals) == 0 and ask % grid_step(ask, figures, decimals) == 0 for bid, ask in quotes)
    mids = [(bid + ask) / 2 for bid, ask in quotes]
    returns = np.array([float(after / before - 1) for before, after in zip(mids, mids[1:])])
    bound = dual_bound(np.array([returns.sum()]), np.array([(returns**2).sum()]), np.array([(returns**3).sum()]))
    bar = max(math.exp(bound), float(mids[-1] / mids[0]))
    bid, price = quotes[-1][0], mids[-1]
    return ledger.wealth({"A": price}), ledger.wealth_at_bid({"A": bid}, {"A": price}), bar, bid


def round_down(value: Fraction, bits: int = 128) -> Fraction:
    """`value` > 0 rounded down to `bits` significant bits: round 4's contract for the confirmer's stored numbers."""
    exponent = value.numerator.bit_length() - value.denominator.bit_length() - bits
    unit = Fraction(2) ** exponent
    return (value // unit) * unit


def ledger_fuzz(runs: int = 300, events: int = 400, seed: int = 41, fee_top: int = 100) -> tuple[int, int, Fraction]:
    """Random long-only accounts on three coins under random stake rules, fills at prices no better than the tape (T1), fee rates from 0 to below
    `fee_top` of 10^-5 (kernel A8 admits 0 to 1), funding of either sign at most a unit's price (V16). Asserts what law K1 says: scales, cash and units never negative, every number on its grid, and no fill
    raising the wealth. Returns the events folded, the ledgers a payment ended, and the largest cash in grid units."""
    rng = np.random.default_rng(seed)
    cents = Fraction(1, 10**8)
    folded = ended = 0
    widest = Fraction(0)
    for _ in range(runs):
        ledger = TestLedger(Fraction(int(rng.integers(1, 11)), 10), Fraction(int(rng.integers(50, 5000))))
        prices = {coin: Fraction(int(rng.integers(10**6, 10**12))) * cents for coin in "ABC"}
        position = {coin: Fraction(0) for coin in "ABC"}
        for _ in range(events):
            coin = "ABC"[int(rng.integers(3))]
            kind = int(rng.integers(4))
            fee = Fraction(int(rng.integers(0, fee_top)), 10**5)
            if kind == 0:
                prices[coin] = max(cents, prices[coin] * Fraction(int(rng.integers(900, 1101)), 1000) // (cents / 2) * (cents / 2))
            elif kind == 1:
                unit_payment = prices[coin] * Fraction(int(rng.integers(-40, 41)), 1000) // (CASH_GRID / TEST_LOT) * (CASH_GRID / TEST_LOT)
                ledger.fund(coin, unit_payment)
            elif kind == 2:
                before = ledger.wealth(prices)
                lots = Fraction(int(rng.integers(1, 500)), 100)
                fill = (prices[coin] * Fraction(int(rng.integers(1000, 1011)), 1000)) // cents * cents + cents
                ledger.buy(coin, lots, fill, fee)
                position[coin] += lots
                assert ledger.scale[coin] >= 0 and ledger.wealth(prices) <= before
            elif position[coin] > 0:
                before = ledger.wealth(prices)
                sold = position[coin] if rng.random() < 0.4 else Fraction(int(rng.integers(1, 100)), 100) * position[coin]
                fill = max(cents, (prices[coin] * Fraction(int(rng.integers(990, 1001)), 1000)) // cents * cents)
                ledger.sell(coin, sold / position[coin], fill, fee)
                position[coin] -= sold
                assert ledger.wealth(prices) <= before
            if ledger.ended:
                ended += 1
                break
            folded += 1
            assert ledger.cash >= 0 and (ledger.cash / CASH_GRID).denominator == 1
            assert all(held >= 0 and (held / TEST_LOT).denominator == 1 for held in ledger.units.values())
            assert all(position[coin] > 0 or ledger.units.get(coin, 0) == 0 for coin in "ABC")

    return folded, ended, widest


def stale_at(events: list[tuple[int, str]], index: int, gap: int) -> bool:
    """Kernel A3: a coin is stale at an event when its latest quote before that event, in the machine's order, is `gap` or more before the event's time,
    or there is none. `events` is one coin's (time, kind) list in that order, a print before a quote of its millisecond."""
    earlier = [time for time, kind in events[:index] if kind == "quote"]
    return not earlier or events[index][0] - earlier[-1] >= gap


def holding_fuzz(paths: int = 2000, seed: int = 43) -> tuple[int, float]:
    """Fixed holdings of three coins, paid for from the starting cash and never traded, on random records of price moves and funding payments of either
    sign. A holding whose cash pays every payment is, exactly, a mean of cash and of each coin held alone with its funding paid in cash, so it is at
    most the best of those. Returns the solvent holdings checked and the largest ratio of one to the best constant mix an optimizer found."""
    rng = np.random.default_rng(seed)
    checked, widest = 0, 0.0
    for _ in range(paths):
        events = 40
        coin_of = rng.integers(3, size=events)
        is_funding = rng.random(events) < 0.4
        size = np.where(is_funding, rng.uniform(-0.3, 0.3, events), rng.uniform(-0.3, 0.5, events))
        weights = rng.dirichlet(np.ones(4))
        price, paid = [Fraction(1)] * 3, [Fraction(0)] * 3
        units = [Fraction(float(weight)) for weight in weights[:3]]
        cash = 1 - sum(units)
        solvent = True
        returns = np.zeros((events, 3))
        for step in range(events):
            coin, amount = int(coin_of[step]), Fraction(float(size[step]))
            if is_funding[step]:
                payment = amount * price[coin]
                paid[coin] += payment
                cash -= units[coin] * payment
                solvent = solvent and cash >= 0
                returns[step, coin] = float(-amount)
            else:
                price[coin] *= 1 + amount
                returns[step, coin] = float(amount)
        if not solvent:
            continue
        checked += 1
        holding = cash + sum(held * now for held, now in zip(units, price))
        alone = [now - funding for now, funding in zip(price, paid)]
        assert holding <= max(Fraction(1), *alone)
        widest = max(widest, float(holding) / math.exp(mix_bound(returns)[0]))
    return checked, widest

WINDOWS = (22, 28, 35, 44, 66)


def sign_path(n_max: int, theta1: float, agent_only: float, searcher_only: float) -> tuple[float, float, float, list[str]]:
    """Study 1: Wald's sequential sign test on discordant markets, theta <= 1/2 against theta1 at nominal errors
    5%, ending undecided at n_max markets, when only the agent's pick earns the plant with chance `agent_only` and
    only G100's with `searcher_only`. Exact, by enumeration over (wins, losses). Returns the chance of the claim, of
    the futility stop, the expected markets, and the chance that the claim, and that either boundary, is reached
    within each count of WINDOWS."""
    upper = math.log(0.95 / 0.05)
    lower = math.log(0.05 / 0.95)
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)
    alive = {(0, 0): 1.0}
    up = down = expected = 0.0
    within = []
    for n in range(1, n_max + 1):
        nxt: dict[tuple[int, int], float] = {}
        for (wins, losses), mass in alive.items():
            steps = ((1, 0, agent_only), (0, 1, searcher_only), (0, 0, 1 - agent_only - searcher_only))
            for more_wins, more_losses, chance in steps:
                state = (wins + more_wins, losses + more_losses)
                ratio = state[0] * win + state[1] * loss
                reached = mass * chance
                if ratio >= upper:
                    up += reached
                    expected += n * reached
                elif ratio <= lower:
                    down += reached
                    expected += n * reached
                elif n == n_max:
                    expected += n * reached
                else:
                    nxt[state] = nxt.get(state, 0.0) + reached
        alive = nxt
        if n in WINDOWS:
            within.append(f"{up:.2f}/{up + down:.2f}")
    return up, down, expected, within


def claim_chance(agent: float, searcher: float) -> float:
    """The chance of study 1's claim when the arms earn independently at these rates."""
    return sign_path(100, 0.7, agent * (1 - searcher), searcher * (1 - agent))[0]


def sign_test(n_max: int, theta1: float, cases: tuple[tuple[str, float, float], ...]) -> None:
    """A case is (label, P(only the agent's pick earns the plant), P(only G100's does))."""
    print(f"  theta1 = {theta1}, at most {n_max} markets")
    for label, agent_only, searcher_only in cases:
        up, down, expected, within = sign_path(n_max, theta1, agent_only, searcher_only)
        theta = agent_only / (agent_only + searcher_only)
        print(f"    {label:44s} theta {theta:.3f}: claim {up:.3f}, futility stop {down:.3f}, undecided {1 - up - down:.3f}, markets {expected:.0f}; within {WINDOWS}: {', '.join(within)}")


def pair_rule() -> tuple[int, int]:
    """A pick earns from the plant when it beats half the reference on at least `majority` of its market's `pairs`
    fresh pairs: a binomial test of q, its chance on one pair. The rule is the smallest that classes a pick at
    q = 1/2 with probability at most 0.005 and a pick at q = 3/4 with probability at least 0.9."""
    found = next(
        (pairs, majority)
        for pairs in range(1, 400)
        for majority in range(pairs + 1)
        if binom.sf(majority - 1, pairs, 0.5) <= 0.005 and binom.sf(majority - 1, pairs, 0.75) >= 0.9
    )
    chances = (0.5, 0.6, 0.671, 0.75, 0.8, 0.9)
    for pairs, majority in ((32, 24), found):
        passes = ", ".join(f"q = {q}: {binom.sf(majority - 1, pairs, q):.4f}" for q in chances)
        print(f"  {majority} of {pairs} pairs, the chance a pick is classed as earning: {passes}")
    print(f"  the rule is {found[1]} of {found[0]}; with exactly opposite effects a program and its mirror win disjoint pairs, and 2 * {found[1]} > {found[0]}, so they never both earn")
    return found


def walls(pairs: int, majority: int) -> None:
    """Wall 3 classifies the even part of each of 5 mirror pairs on 200 markets, each pair against its own count.
    Wall 5 holds a program that reads no input below the bottom of G100's band. Argon's momentum: the share of the
    reference's Sharpe that an average of past returns keeps, when the plant is an AR(1) drift."""
    size = binom.sf(majority - 1, pairs, 0.5)
    mirror_pairs, markets = 5, 200

    def any_pair_stops(count: int) -> float:
        return float(1 - (1 - binom.sf(count, markets, size)) ** mirror_pairs)

    allowed = next(count for count in range(markets) if any_pair_stops(count) <= 0.01)
    print(f"  wall 3: an even part at q <= 1/2 earns with probability at most {size:.4f}; more than {allowed} of {markets} for some of the {mirror_pairs} mirror pairs has probability {any_pair_stops(allowed):.4f} (at {allowed - 1}: {any_pair_stops(allowed - 1):.4f}); an even part that earns on 4.5% of markets passes with probability {binom.cdf(allowed, markets, 0.045):.3f}")
    bottom = 0.11
    constant = max(count for count in range(markets) if binom.cdf(count, markets, bottom) <= 0.05)

def reruns(n_max: int = 100, theta1: float = 0.7) -> None:
    """Why study 1 runs no session again, at the worst: every market discordant at one half, and a rerun that hides a
    loss whenever hiding one raises the chance of the claim. Exact, by backward induction over (wins, losses,
    markets, reruns)."""
    upper = math.log(19)
    win = math.log(theta1 / 0.5)
    loss = math.log((1 - theta1) / 0.5)

    @functools.cache
    def claim(wins: int, losses: int, markets: int, left: int) -> float:
        ratio = wins * win + losses * loss
        if ratio >= upper:
            return 1.0
        if ratio <= -upper or markets == n_max:
            return 0.0
        kept = claim(wins, losses + 1, markets + 1, left)
        hidden = claim(wins, losses, markets, left - 1) if left else 0.0
        return 0.5 * claim(wins + 1, losses, markets + 1, left) + 0.5 * max(kept, hidden)

    sizes = ", ".join(f"{count}: {claim(0, 0, 0, count):.3f}" for count in range(6))
    print(f"  the size when up to v reruns each hide a loss, by v: {sizes}")
    for share in (19 / 40, 89 / 128):
        print(f"  were a session with a fallback run again (a share of {share:.3f}), a counted market would cost {1 / (1 - share):.1f} sessions: {29 / (1 - share):.0f} for 29 markets, {41 / (1 - share):.0f} for 41")


def calibration(pairs: int) -> None:
    """The strength is where G100's rate on `search` markets is within 0.005 of 0.15; it is accepted when the rate
    on `check` fresh markets is within `tolerance` of 0.15. A false stop is that check failing by sampling alone."""
    target = 0.15 * 0.85

    def false_stop(search: int, check: int, tolerance: float, offset: float) -> float:
        spread = math.sqrt(target / search + target / check)
        return float(norm.sf((tolerance - offset) / spread) + norm.sf((tolerance + offset) / spread))

    print(f"  revision 4, a search on 400 markets and a check on 2,000 within 0.02: a false stop with probability {false_stop(400, 2000, 0.02, 0):.2f}")
    search, check, tolerance = 1600, 10_000, 0.03
    edge = math.sqrt(0.19 * 0.81 / check)
    print(f"  a search on {search:,} and a check on {check:,} within {tolerance}: a false stop with probability {false_stop(search, check, tolerance, 0.005):.4f}; an accepted strength has a true rate within {tolerance + 2.5 * edge:.3f} of 0.15 (2.5 standard errors of {edge:.4f} at 0.19)")
    market = 100 + 2 * 2 * pairs
    steps = (9, 13)
    wall3 = 200 * 2 * pairs * (10 + 1)
    print(f"  kernel evaluations: {3 * 2 * pairs} a study market to classify both picks and the reference, {100 * 3 * 2 * pairs:,} for 100; a calibration market at one strength, {market}; the search at {steps[0]} strengths (at most {steps[1]}) and the check: {(steps[0] * search + check) * market:,} (at most {(steps[1] * search + check) * market:,}); wall 3: {wall3:,}")
    calibration_cost(pairs, search, check, steps)


def calibration_cost(pairs: int, search: int, check: int, steps: tuple[int, int]) -> None:
    """What the calibration costs in seconds of one core, from the parts of one market. Measured 2026-11-13 on this
    PC at a3fc471, means over 8 markets: `generate`; `gate.markets.write`, which hashes each of a market's 262,812
    rows in Python (ingest/store.py); the same market's search period written with the columns the kernel reads;
    one `mm eval` process, which loads the market; one locked program in it; G100's 100 evaluations. The hosts are
    run `cal-2`'s, 2026-11-15: each one's workers and the markets an hour it finished in round `r0`, all of them
    busy."""
    generate, start, program, g100 = 0.095, 0.18, 0.05, 0.88
    stored, scored = (1.71, 13.3e6), (0.05, 2.9e6)
    blind_markets, mirror_set, cores = 200, 10, 30

    def hours(write: float, twins_once: bool, strengths: int) -> float:
        one = generate + write + start + 2 * program
        searched = generate + stored[0] + g100
        planted = searched + pairs * one
        twins = pairs * one
        search_seconds = search * (strengths * planted + (1 if twins_once else strengths) * twins)
        blind = blind_markets * pairs * 2 * (generate + write + start + (mirror_set + 1) * program)
        return (search_seconds + check * (planted + twins) + blind) / 3600

    pair = 2 * (generate + scored[0] + start + 2 * program)
    print(f"  one fresh pair, generated, written and scored for a pick and the reference: {2 * (generate + stored[0] + start + 2 * program):.2f} s through the revision store, {pair:.2f} s with the columns the kernel reads; a calibration market at one strength, G100's search and its {pairs} pairs: {generate + stored[0] + g100 + pairs * pair:.0f} s")
    for name, write, twins_once in (("as built, each search twin written once and kept", stored[0], True), ("each pair written for scoring, nothing kept", scored[0], False)):
        low, high = (hours(write, twins_once, count) for count in steps)
        print(f"  {name}: {low:,.0f} core-hours at {steps[0]} strengths, {high:,.0f} at {steps[1]}; on {cores} cores {low / cores:.1f} to {high / cores:.1f} hours, on 26 {low / 26:.1f} to {high / 26:.1f}, on 12 {low / 12:.1f} to {high / 12:.1f}")
    print(f"  disk: the search's twins kept through the revision store, {search * pairs * stored[1] / 1e12:.2f} TB; with nothing kept, a worker holds one searched market or one pair, at most {max(stored[1], 2 * scored[1]) / 1e6:.1f} MB, so {cores} workers hold {cores * max(stored[1], 2 * scored[1]) / 1e9:.2f} GB")
    hosts = {"this PC": (11, 279), "prodesk7": (10, 164), "gmktec": (6, 59)}
    workers = sum(count for count, _ in hosts.values())
    together = sum(rate for _, rate in hosts.values())
    blind_equivalent = math.ceil(blind_markets * (mirror_set + 1) * 2 * pairs / (100 + 2 * 2 * pairs))
    work = [count * search + check + blind_equivalent for count in (1, *steps)]
    by_rate = [markets / together for markets in work]
    by_cores = [max(markets * count / workers / rate for count, rate in hosts.values()) for markets in work]
    print(f"  as measured on the hosts that run it ({', '.join(f'{name}: {rate} markets an hour on {count} workers' for name, (count, rate) in hosts.items())}): {together} an hour together, {3600 / together:.2f} s a market")
    print(f"    a round of the search split by rates: shares {', '.join(str(round(search * rate / together)) for _, rate in hosts.values())}, {search / together:.1f} h; by workers: shares {', '.join(str(round(search * count / workers)) for count, _ in hosts.values())}, {max(search * count / workers / rate for count, rate in hosts.values()):.1f} h")
    print(f"    the job at one strength, {steps[0]} and {steps[1]}: {', '.join(f'{h:.1f}' for h in by_rate)} h by rates; {', '.join(f'{h:.1f}' for h in by_cores)} h by workers")
    for name, (count, rate) in hosts.items():
        part = rate / together
        print(f"    {name}: {part:.1%} of the rate; without it a round by rates is {search / (together - rate):.1f} h; it shortens the job while its weight's error is below {part / (1 - part):.0%}; by workers its error is {count / workers / part - 1:+.0%}")
    bests = 5
    print(f"  the study's arms: G100's pick, its {bests} earlier bests and the reference on a market's {pairs} pairs, {pairs * 2 * (generate + scored[0] + start + (bests + 2) * program):.0f} s; the agent's pick and the reference, {pairs * pair:.0f} s; 100 markets, both arms, {100 * pairs * (2 * (generate + scored[0] + start + (bests + 2) * program) + pair) / 3600:.1f} core-hours")

def expected_max(count: int) -> float:
    """The expected maximum of `count` independent standard Gaussians, by quadrature."""
    value, _ = quad(lambda x: x * count * norm.pdf(x) * norm.cdf(x) ** (count - 1), -10, 12, limit=200)
    return float(value)


def dsr_z(sharpe: float, threshold: float, steps: int, skew: float, kurtosis: float) -> float:
    spread = math.sqrt(1 - skew * sharpe + (kurtosis - 1) / 4 * sharpe**2)
    return (sharpe - threshold) * math.sqrt(steps - 1) / spread

def cut(whole: int, factor: float) -> int:
    """The venue's scaling of a whole number: the double product, its fraction dropped (facts section 7)."""
    return math.trunc(whole * factor)


def venue_fee(notional: int, rate: Fraction) -> int:
    return cut(notional, float(rate))


def venue_close(entry_notional: int, lots: int, closed: int, price: int) -> tuple[int, int]:
    """A close's printed closedPnl and the entry notional it leaves, all in millionths of a USDC and price units of 10^(szDecimals - 6)."""
    printed = closed * (price - entry_notional // lots)
    left = lots - closed
    kept = cut(entry_notional, left / lots) if left else 0
    return printed, kept


def whole(text: str, decimals: int) -> int:
    scale: int = 10**decimals
    scaled = Fraction(text) * scale
    assert scaled.denominator == 1, (text, decimals)
    return scaled.numerator


def venue_prints(fills_directory: Path | None) -> None:
    """The venue's printed fee and closedPnl. Three made-up cases always; given the directory of the vault's fills pages (dated before
    2026-11-01, untracked), counts over the whole record, and never a price."""
    maker, taker = Fraction(15, 100_000), Fraction(45, 100_000)
    assert venue_fee(100_000, maker) == 14 and Fraction(100_000) * maker == 15
    assert venue_fee(100_000, taker) == 45 and venue_fee(33_333, maker) == 4
    assert venue_close(302_000_000, 3, 3, 102_000_000) == (4_000_002, 0)
    first, kept = venue_close(13_574, 11, 8, 2_000)
    assert (first, kept) == (6_128, 3_701) and 13_574 * 3 % 11 == 0 and 13_574 * 3 // 11 == 3_702
    assert venue_close(kept, 3, 3, 2_000) == (2_301, 0)
    print("  made-up cases hold: a maker fee of exactly 15 millionths prints 14; one lot at 100 and two at 101 closed at 102 print 4.000002;")
    print("    11 lots at 1,234 units, 8 closed, leave an entry notional of 3,701 where the exact is 3,702, and the last 3 close one unit low")
    if fills_directory is None:
        print("  the vault's record: not given (pass the directory of its fills pages)")
        return
    universe = json.loads(Path("venue/hyperliquid/meta-2026-11-16.json").read_text())["universe"]
    size_decimals = {coin["name"]: coin["szDecimals"] for coin in universe}
    fills: list[dict[str, object]] = []
    seen: set[int] = set()
    for page in sorted(fills_directory.glob("fills-*.json")):
        for row in json.loads(page.read_text()):
            if row["tid"] not in seen:
                seen.add(row["tid"])
                fills.append(row)
    fills.sort(key=lambda row: int(str(row["time"])))
    lots: dict[str, int] = {}
    entry: dict[str, int] = {}
    cost: dict[str, Fraction] = {}
    partials: dict[str, int] = {}
    fee_differs = fee_whole = fee_one_under = fee_outside = 0
    closes = pnl_differs = clean_closes = clean_equal = bound_broken = below_exact = one_unit_low = partial_closes = lossy = 0
    printed_sum = exact_sum = largest = Fraction(0)
    for row in fills:
        coin = str(row["coin"])
        decimals = size_decimals[coin]
        price, size = whole(str(row["px"]), 6 - decimals), whole(str(row["sz"]), decimals)
        held = lots.get(coin, 0)
        assert whole(str(row["startPosition"]), decimals) == held, "the position chain breaks"
        notional = price * size
        rate = taker if row["crossed"] else maker
        printed_fee = whole(str(row["fee"]), 6)
        exact_fee = notional * rate
        fee_differs += venue_fee(notional, rate) != printed_fee
        fee_outside += not 0 <= exact_fee - printed_fee <= 1
        fee_whole += exact_fee.denominator == 1
        fee_one_under += exact_fee - printed_fee == 1
        if row["side"] == "B":
            if held == 0:
                entry[coin], cost[coin], partials[coin] = 0, Fraction(0), 0
            entry[coin] += notional
            cost[coin] += notional
            lots[coin] = held + size
            continue
        closes += 1
        printed = whole(str(row["closedPnl"]), 6)
        ours, kept = venue_close(entry[coin], held, size, price)
        exact = size * (price - cost[coin] / held)
        pnl_differs += ours != printed
        if partials[coin] == 0:
            clean_closes += 1
            clean_equal += entry[coin] == cost[coin]
        bound_broken += not cost[coin] - partials[coin] <= entry[coin] <= cost[coin]
        below_exact += printed < exact
        one_unit_low += entry[coin] // held < math.floor(cost[coin] / held)
        printed_sum += printed
        exact_sum += exact
        largest = max(largest, printed - exact)
        left = held - size
        if left:
            partial_closes += 1
            partials[coin] += 1
            lossy += kept != entry[coin] * left // held
        entry[coin], cost[coin], lots[coin] = kept, cost[coin] * left / held, left
    print(f"  {len(fills)} fills: {fee_differs} printed fees differ from the cut; {fee_outside} outside [exact - 1, exact] millionths; {fee_whole} exact fees are whole, {fee_one_under} print one under")
    print(f"  {closes} closes: {pnl_differs} printed closedPnl differ; {clean_equal} of {clean_closes} before any partial close have the entry notional equal to the exact cost;")
    print(f"    {bound_broken} break cost - partial closes <= entry notional <= cost; {below_exact} print below the exact figure; {one_unit_low} have an entry one unit under the exact floor")
    print(f"    {partial_closes} partial closes, {lossy} leave the entry notional under its exact floor")
    print(f"    printed {float(printed_sum) / 1e6:.6f} USDC against {float(exact_sum) / 1e6:.6f} exact, {float(printed_sum - exact_sum) / 1e6:.6f} over; the most at one close {float(largest) / 1e6:.4f}")

    for page in sorted(vault_directory.glob("funding-*.json")):
        for row in json.loads(page.read_text()):
            rows[int(row["time"]), str(row["delta"]["coin"])] = row["delta"]
    path: dict[str, list[tuple[int, Fraction]]] = {}
    position: dict[str, Fraction] = {}
    for fill in sorted(fills.values(), key=lambda fill: int(str(fill["time"]))):
        coin = str(fill["coin"])
        signed = Fraction(str(fill["sz"])) * (1 if fill["side"] == "B" else -1)
        position[coin] = position.get(coin, Fraction(0)) + signed
        path.setdefault(coin, []).append((int(str(fill["time"])), position[coin]))

    def held_into(coin: str, time: int) -> Fraction:
        held = Fraction(0)
        for at, after in path[coin]:
            if at >= time:
                break
            held = after
        return held

    def held_hours(coin: str, day: int) -> list[tuple[int, Fraction]]:
        hours = [(day + index * hour, held_into(coin, day + index * hour)) for index in range(24)]
        return [(at, size) for at, size in hours if size != 0]

    first, last = min(day for day, _ in rows) - 24 * hour, max(day for day, _ in rows)
    held_days = {(day, coin) for coin in path for day in range(first, last + 1, 24 * hour) if held_hours(coin, day)}
    count_differs = size_differs = 0
    for (day, coin), row in rows.items():
        held = held_hours(coin, day)
        count_differs += len(held) != row["nSamples"]
        size_differs += not held or abs(sum(size for _, size in held) / len(held) - Fraction(str(row["szi"]))) > unit / 2
    paid = [Fraction(str(row["usdc"])) for row in rows.values()]
    print(f"  {len(rows)} rows, every time a UTC midnight: {all(day % (24 * hour) == 0 for day, _ in rows)}; {len(held_days - set(rows))} held coin-days have no row and {len(set(rows) - held_days)} rows no held hour")
    print(f"    {count_differs} differ in nSamples from the hours held of the day that starts at the row's time; {size_differs} in szi from the mean size held, at half of 1e-10")
    print(f"    usdc sums to {float(sum(paid)):.2f} received, {float(sum(abs(cash) for cash in paid)):.2f} gross")
    if rates_file is None:
        print("  fundingHistory rates: not given")
        return
    rate: dict[tuple[str, int], Fraction] = {}
    stamps: list[int] = []
    premiums: list[float] = []
    for coin, history in json.loads(rates_file.read_text()).items():
        for settled in history:
            rate[coin, int(settled["time"]) // hour * hour] = Fraction(settled["fundingRate"])
            stamps.append(int(settled["time"]) % hour)
            premiums.append(abs(float(settled["premium"])))
    covered = rate_differs = one_signed = outside = 0
    for (day, coin), row in rows.items():
        held = held_hours(coin, day)
        if not all((coin, at) in rate for at, _ in held):
            continue
        covered += 1
        weight = sum((size * rate[coin, at] for at, size in held), Fraction(0))
        rate_differs += abs(weight / sum(size for _, size in held) - Fraction(str(row["fundingRate"]))) > unit / 2
        signs = {rate[coin, at] > 0 for at, _ in held}
        if weight != 0 and len(signs) == 1:
            one_signed += 1
            implied = -Fraction(str(row["usdc"])) / weight
            prices = [Fraction(str(fill["px"])) for fill in fills.values() if fill["coin"] == coin and abs(int(str(fill["time"])) - day - 12 * hour) <= 36 * hour]
            outside += not min(prices) * Fraction(97, 100) <= implied <= max(prices) * Fraction(103, 100)
    print(f"  {covered} rows have every held hour's rate: {rate_differs} differ in fundingRate from the size-weighted mean rate, at half of 1e-10")
    print(f"    {one_signed} of them have rates of one sign: {outside} imply a price, -usdc over the sum of size times rate, outside the vault's fills of the three days around, widened 3%")
    print(f"    stamps at most {max(stamps)} ms after their hour; |premium| median {statistics.median(premiums):.5f}, greatest {max(premiums):.5f}")
